## 22. 请手写快速排序

**题目：** 手写实现快速排序算法。

**参考答案：**

```javascript
// 方法1：经典快速排序（使用额外空间）
function quickSort(arr) {
  if (arr.length <= 1) {
    return arr;
  }

  const pivot = arr[Math.floor(arr.length / 2)];
  const left = [];
  const right = [];
  const equal = [];

  for (const item of arr) {
    if (item < pivot) {
      left.push(item);
    } else if (item > pivot) {
      right.push(item);
    } else {
      equal.push(item);
    }
  }

  return [...quickSort(left), ...equal, ...quickSort(right)];
}

// 方法2：原地快速排序
function quickSortInPlace(arr, left = 0, right = arr.length - 1) {
  if (left >= right) {
    return;
  }

  const pivotIndex = partition(arr, left, right);
  quickSortInPlace(arr, left, pivotIndex - 1);
  quickSortInPlace(arr, pivotIndex + 1, right);
}

function partition(arr, left, right) {
  const pivot = arr[right];
  let i = left;

  for (let j = left; j < right; j++) {
    if (arr[j] < pivot) {
      [arr[i], arr[j]] = [arr[j], arr[i]];
      i++;
    }
  }

  [arr[i], arr[right]] = [arr[right], arr[i]];
  return i;
}

// 方法3：三路快速排序（处理大量重复元素）
function quickSort3Way(arr, left = 0, right = arr.length - 1) {
  if (left >= right) {
    return;
  }

  let lt = left;    // 小于pivot的右边界
  let gt = right;   // 大于pivot的左边界
  let i = left + 1; // 当前元素
  const pivot = arr[left];

  while (i <= gt) {
    if (arr[i] < pivot) {
      [arr[i], arr[lt]] = [arr[lt], arr[i]];
      lt++;
      i++;
    } else if (arr[i] > pivot) {
      [arr[i], arr[gt]] = [arr[gt], arr[i]];
      gt--;
    } else {
      i++;
    }
  }

  quickSort3Way(arr, left, lt - 1);
  quickSort3Way(arr, gt + 1, right);
}

// 测试
const arr1 = [3, 6, 8, 10, 1, 2, 1];
console.log('Original:', arr1);
console.log('QuickSort:', quickSort([...arr1])); // [1, 1, 2, 3, 6, 8, 10]

const arr2 = [3, 6, 8, 10, 1, 2, 1];
quickSortInPlace(arr2);
console.log('QuickSort In Place:', arr2); // [1, 1, 2, 3, 6, 8, 10]

const arr3 = [3, 3, 1, 2, 3, 2, 1, 3];
quickSort3Way(arr3);
console.log('QuickSort 3 Way:', arr3); // [1, 1, 2, 2, 3, 3, 3, 3]
```
