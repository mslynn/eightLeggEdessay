## 14. 去除字符串中出现次数最少的字符，不改变原字符串的顺序

**题目：** 去除字符串中出现次数最少的字符，不改变原字符串的顺序。

**参考答案：**

```javascript
function removeLeastFrequentChars(str) {
  if (!str) {
    return '';
  }

  // 统计每个字符出现的次数
  const charCount = {};
  for (const char of str) {
    charCount[char] = (charCount[char] || 0) + 1;
  }

  // 找出出现次数最少
  const minCount = Math.min(...Object.values(charCount));

  // 去除出现次数最少的字符
  let result = '';
  for (const char of str) {
    if (charCount[char] > minCount) {
      result += char;
    }
  }

  return result;
}

// 测试
console.log(removeLeastFrequentChars('aabcccccaaa')); // b1 (因为b只出现1次，a和c出现多次)
console.log(removeLeastFrequentChars('hello world')); // heo word (l出现3次最多，o出现2次，h、e、w、r、d各出现1次)
console.log(removeLeastFrequentChars('abc')); // '' (每个字符都出现1次)
console.log(removeLeastFrequentChars('aaabbbcc')); // aaabbbcc (每个字符出现次数相同)
console.log(removeLeastFrequentChars('aabccc')); // ccc (b出现1次最少)
```
