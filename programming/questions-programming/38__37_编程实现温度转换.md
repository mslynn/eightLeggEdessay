## 37. 编程实现温度转换

**题目：** 已知温度转换的关系式是：华氏度＝32＋摄氏度×1.8，现在要求输入摄氏度，输出对应的华氏度，小数保留两位。

**参考答案：**

```javascript
function celsiusToFahrenheit(celsius) {
  const fahrenheit = 32 + celsius * 1.8;
  return Number(fahrenheit.toFixed(2));
}

// 测试
console.log(celsiusToFahrenheit(0));   // 32
console.log(celsiusToFahrenheit(100)); // 212
console.log(celsiusToFahrenheit(37));  // 98.6
console.log(celsiusToFahrenheit(-40)); // -40

// 华氏度转摄氏度
function fahrenheitToCelsius(fahrenheit) {
  const celsius = (fahrenheit - 32) / 1.8;
  return Number(celsius.toFixed(2));
}

// 测试
console.log(fahrenheitToCelsius(32));   // 0
console.log(fahrenheitToCelsius(212));  // 100
console.log(fahrenheitToCelsius(98.6)); // 37
console.log(fahrenheitToCelsius(-40));  // -40

// 通用温度转换函数
function convertTemperature(value, from, to) {
  const fromLower = from.toLowerCase();
  const toLower = to.toLowerCase();

  // 先转换为摄氏度
  let celsius;
  if (fromLower === 'c' || fromLower === 'celsius') {
    celsius = value;
  } else if (fromLower === 'f' || fromLower === 'fahrenheit') {
    celsius = (value - 32) / 1.8;
  } else if (fromLower === 'k' || fromLower === 'kelvin') {
    celsius = value - 273.15;
  } else {
    throw new Error(`Unknown temperature unit: ${from}`);
  }

  // 从摄氏度转换为目标单位
  if (toLower === 'c' || toLower === 'celsius') {
    return Number(celsius.toFixed(2));
  } else if (toLower === 'f' || toLower === 'fahrenheit') {
    return Number((32 + celsius * 1.8).toFixed(2));
  } else if (toLower === 'k' || toLower === 'kelvin') {
    return Number((celsius + 273.15).toFixed(2));
  } else {
    throw new Error(`Unknown temperature unit: ${to}`);
  }
}

// 测试
console.log(convertTemperature(37, 'celsius', 'fahrenheit')); // 98.6
console.log(convertTemperature(98.6, 'fahrenheit', 'celsius')); // 37
console.log(convertTemperature(273.15, 'kelvin', 'celsius')); // 0
console.log(convertTemperature(0, 'celsius', 'kelvin')); // 273.15
```
