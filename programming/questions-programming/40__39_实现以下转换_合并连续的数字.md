## 39. 实现以下转换，合并连续的数字

**题目：** 将 `[1,2,3,4,6,7,9,13,15]` 转换为 `['1->4','6->7','9','13','15']`

**参考答案：**

```javascript
function mergeContinuousNumbers(nums) {
  if (nums.length === 0) {
    return [];
  }

  const result = [];
  let start = nums[0];
  let end = nums[0];

  for (let i = 1; i < nums.length; i++) {
    if (nums[i] === end + 1) {
      // 连续
      end = nums[i];
    } else {
      // 不连续，记录上一段
      if (start === end) {
        result.push(`${start}`);
      } else {
        result.push(`${start}->${end}`);
      }
      start = nums[i];
      end = nums[i];
    }
  }

  // 记录最后一段
  if (start === end) {
    result.push(`${start}`);
  } else {
    result.push(`${start}->${end}`);
  }

  return result;
}

// 测试
console.log(mergeContinuousNumbers([1, 2, 3, 4, 6, 7, 9, 13, 15]));
// ['1->4', '6->7', '9', '13', '15']

console.log(mergeContinuousNumbers([1, 2, 3]));
// ['1->3']

console.log(mergeContinuousNumbers([1]));
// ['1']

console.log(mergeContinuousNumbers([]));
// []

console.log(mergeContinuousNumbers([1, 3, 5]));
// ['1', '3', '5']

// 反向转换：从字符串范围转回数组
function expandRanges(ranges) {
  const result = [];

  for (const range of ranges) {
    if (range.includes('->')) {
      const [start, end] = range.split('->').map(Number);
      for (let i = start; i <= end; i++) {
        result.push(i);
      }
    } else {
      result.push(Number(range));
    }
  }

  return result;
}

// 测试
console.log(expandRanges(['1->4', '6->7', '9', '13', '15']));
// [1, 2, 3, 4, 6, 7, 9, 13, 15]
```
