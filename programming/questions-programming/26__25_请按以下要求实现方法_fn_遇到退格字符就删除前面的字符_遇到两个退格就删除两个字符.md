## 25. 请按以下要求实现方法 fn：遇到退格字符就删除前面的字符，遇到两个退格就删除两个字符

**题目：** 实现一个函数，处理字符串中的退格字符（用<表示），一个<删除前一个字符，两个<<删除前两个字符。

**参考答案：**

```javascript
function processBackspaces(str) {
  const stack = [];
  let backspaceCount = 0;

  for (let i = str.length - 1; i >= 0; i--) {
    const char = str[i];

    if (char === '<') {
      backspaceCount++;
    } else if (backspaceCount > 0) {
      backspaceCount--;
    } else {
      stack.unshift(char);
    }
  }

  return stack.join('');
}

// 方法2：使用栈（从前往后处理）
function processBackspaces2(str) {
  const stack = [];

  for (const char of str) {
    if (char === '<') {
      stack.pop();
    } else {
      stack.push(char);
    }
  }

  return stack.join('');
}

// 测试
console.log(processBackspaces('ab<c')); // a
console.log(processBackspaces('ab<<c')); // c
console.log(processBackspaces('a<bc<d')); // bd
console.log(processBackspaces('abc<<<<d')); // d
console.log(processBackspaces('<<ab<c')); // （删除两个字符后再删除一个）
console.log(processBackspaces2('ab<c')); // a
console.log(processBackspaces2('ab<<c')); // c
console.log(processBackspaces2('a<bc<d')); // bd
console.log(processBackspaces2('abc<<<<d')); // d
```
