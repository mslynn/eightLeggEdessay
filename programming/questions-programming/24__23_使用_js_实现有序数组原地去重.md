## 23. 使用 js 实现有序数组原地去重

**题目：** 对有序数组进行原地去重。

**参考答案：**

```javascript
// 方法1：双指针
function removeDuplicates(nums) {
  if (nums.length <= 1) {
    return nums.length;
  }

  let slow = 0; // 慢指针，指向不重复的位置

  for (let fast = 1; fast < nums.length; fast++) {
    if (nums[fast] !== nums[slow]) {
      slow++;
      nums[slow] = nums[fast];
    }
  }

  return slow + 1;
}

// 方法2：使用Set（不是原地，但更简洁）
function removeDuplicates2(nums) {
  const unique = [...new Set(nums)];
  nums.length = 0;
  nums.push(...unique);
  return nums.length;
}

// 测试
const arr1 = [1, 1, 2, 2, 3, 4, 4, 5];
const length1 = removeDuplicates(arr1);
console.log('去重后数组:', arr1.slice(0, length1)); // [1, 2, 3, 4, 5]
console.log('新长度:', length1); // 5

const arr2 = [1, 1, 1, 2, 2, 3];
const length2 = removeDuplicates(arr2);
console.log('去重后数组:', arr2.slice(0, length2)); // [1, 2, 3]
console.log('新长度:', length2); // 3

const arr3 = [];
const length3 = removeDuplicates(arr3);
console.log('空数组去重:', arr3.slice(0, length3)); // []
console.log('新长度:', length3); // 0
```
