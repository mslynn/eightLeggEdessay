## 30. 实现一个方法，从某个数值数组中，获取最小正数（非零非负数）的索引值

**题目：** 从数值数组中获取最小正数（非零非负数）的索引。

**参考答案：**

```javascript
function findMinPositiveIndex(arr) {
  let minValue = Infinity;
  let minIndex = -1;

  for (let i = 0; i < arr.length; i++) {
    if (arr[i] > 0 && arr[i] < minValue) {
      minValue = arr[i];
      minIndex = i;
    }
  }

  return minIndex;
}

// 方法2：使用reduce
function findMinPositiveIndex2(arr) {
  let result = { value: Infinity, index: -1 };

  arr.forEach((num, index) => {
    if (num > 0 && num < result.value) {
      result = { value: num, index };
    }
  });

  return result.index;
}

// 方法3：使用filter和indexOf
function findMinPositiveIndex3(arr) {
  const positiveNumbers = arr.filter(num => num > 0);

  if (positiveNumbers.length === 0) {
    return -1;
  }

  const minPositive = Math.min(...positiveNumbers);
  return arr.indexOf(minPositive);
}

// 测试
const arr1 = [0, -1, 2, 3, 1, -5, 4];
console.log(findMinPositiveIndex(arr1)); // 4 (值为1)
console.log(findMinPositiveIndex2(arr1)); // 4
console.log(findMinPositiveIndex3(arr1)); // 4

const arr2 = [-1, -2, -3];
console.log(findMinPositiveIndex(arr2)); // -1
console.log(findMinPositiveIndex2(arr2)); // -1
console.log(findMinPositiveIndex3(arr2)); // -1

const arr3 = [5, 3, 7, 1, 9, 2];
console.log(findMinPositiveIndex(arr3)); // 3 (值为1)
```
