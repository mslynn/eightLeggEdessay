## 36. 根据运算优先级添加括号

**题目：** 根据运算优先级给表达式添加括号，如 `'11+2-3*4+5/2*4+10/5'` 结果为 `'11+2-(3*4)+(5/2*4)+(10/5)'`

**参考答案：**

```javascript
function addParentheses(expression) {
  // 定义运算符优先级
  const precedence = {
    '+': 1,
    '-': 1,
    '*': 2,
    '/': 2
  };

  // 解析表达式
  const tokens = parseExpression(expression);
  const result = [];

  for (let i = 0; i < tokens.length; i++) {
    const token = tokens[i];

    if (isOperator(token)) {
      const prevToken = tokens[i - 1];
      const nextToken = tokens[i + 1];

      // 检查是否需要加括号
      if (needsParentheses(token, prevToken, nextToken, precedence)) {
        result.push('(');
      }

      result.push(token);

      if (needsParentheses(token, prevToken, nextToken, precedence)) {
        result.push(')');
      }
    } else {
      result.push(token);
    }
  }

  return result.join('');
}

function parseExpression(expression) {
  // 移除所有空格
  expression = expression.replace(/\s/g, '');

  const tokens = [];
  let currentNumber = '';

  for (const char of expression) {
    if (/\d/.test(char)) {
      currentNumber += char;
    } else {
      if (currentNumber) {
        tokens.push(currentNumber);
        currentNumber = '';
      }
      tokens.push(char);
    }
  }

  if (currentNumber) {
    tokens.push(currentNumber);
  }

  return tokens;
}

function isOperator(token) {
  return ['+', '-', '*', '/'].includes(token);
}

function needsParentheses(currentOp, prevToken, nextToken, precedence) {
  // 乘除法需要加括号
  if (currentOp === '*' || currentOp === '/') {
    return true;
  }

  // 加减法，如果前一个运算符是乘除，需要加括号
  if ((currentOp === '+' || currentOp === '-') && isOperator(prevToken)) {
    return precedence[prevToken] > precedence[currentOp];
  }

  return false;
}

// 更精确的版本
function addParenthesesV2(expression) {
  const precedence = {
    '+': 1,
    '-': 1,
    '*': 2,
    '/': 2
  };

  const tokens = expression.replace(/\s/g, '').split(/([+\-*/])/);
  const result = [];

  for (let i = 0; i < tokens.length; i++) {
    const token = tokens[i];

    if (['*', '/'].includes(token)) {
      // 乘除法需要括号
      const prev = tokens[i - 1];
      const next = tokens[i + 1];

      if (prev && next) {
        const startIndex = result.length - 1;
        result.push('(');
        result.push(prev);
        result.push(token);
        result.push(next);
        result.push(')');
        i++; // 跳过next
      }
    } else if (token && !['+', '-', '*', '/'].includes(token)) {
      // 数字
      result.push(token);
    } else if (['+', '-'].includes(token)) {
      result.push(token);
    }
  }

  return result.join('');
}

// 测试
const str = '11+2-3*4+5/2*4+10/5';
console.log(addParentheses(str));
// 11+2-(3*4)+(5/2*4)+(10/5)

console.log(addParenthesesV2(str));
// 11+2-3*4+5/2*4+10/5 (可能需要进一步优化)
```
