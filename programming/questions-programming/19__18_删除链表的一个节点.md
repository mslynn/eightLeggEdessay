## 18. 删除链表的一个节点

**题目：** 给定单向链表的头节点和一个要删除的节点的值，删除该节点。

**参考答案：**

```javascript
// 链表节点定义
class ListNode {
  constructor(val) {
    this.val = val;
    this.next = null;
  }
}

// 删除链表节点
function deleteNode(head, val) {
  // 头节点就是要删除的节点
  if (head.val === val) {
    return head.next;
  }

  let current = head;

  while (current.next) {
    if (current.next.val === val) {
      current.next = current.next.next;
      return head;
    }
    current = current.next;
  }

  // 没找到要删除的节点
  return head;
}

// 辅助函数：创建链表
function createLinkedList(arr) {
  if (arr.length === 0) return null;

  const head = new ListNode(arr[0]);
  let current = head;

  for (let i = 1; i < arr.length; i++) {
    current.next = new ListNode(arr[i]);
    current = current.next;
  }

  return head;
}

// 辅助函数：打印链表
function printLinkedList(head) {
  const result = [];
  let current = head;

  while (current) {
    result.push(current.val);
    current = current.next;
  }

  console.log(result.join(' -> '));
}

// 测试
const list = createLinkedList([1, 2, 3, 4, 5]);
console.log('原链表:');
printLinkedList(list); // 1 -> 2 -> 3 -> 4 -> 5

// 删除节点3
const newList = deleteNode(list, 3);
console.log('删除3后的链表:');
printLinkedList(newList); // 1 -> 2 -> 4 -> 5

// 删除头节点
const newList2 = deleteNode(newList, 1);
console.log('删除1后的链表:');
printLinkedList(newList2); // 2 -> 4 -> 5

// 删除不存在的节点
const newList3 = deleteNode(newList2, 10);
console.log('删除10后的链表:');
printLinkedList(newList3); // 2 -> 4 -> 5
```
