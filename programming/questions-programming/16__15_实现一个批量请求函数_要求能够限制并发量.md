## 15. 实现一个批量请求函数，要求能够限制并发量

**题目：** 实现一个批量请求函数，能够限制并发请求数量。

**参考答案：**

### 方案一：RequestPool 类 + 手动等待所有请求（不使用 Promise.all）

```javascript
class RequestPool {
  constructor(concurrency = 5) {
    this.concurrency = concurrency;
    this.queue = [];
    this.activeCount = 0;
  }

  async add(requestFn) {
    return new Promise((resolve, reject) => {
      this.queue.push({
        requestFn,
        resolve,
        reject
      });

      this.runNext();
    });
  }

  async runNext() {
    // 如果正在执行的请求数已达到上限，或者队列为空，则不执行
    if (this.activeCount >= this.concurrency || this.queue.length === 0) {
      return;
    }

    // 从队列中取出一个请求
    const { requestFn, resolve, reject } = this.queue.shift();
    this.activeCount++;

    try {
      const result = await requestFn();
      resolve(result);
    } catch (error) {
      reject(error);
    } finally {
      this.activeCount--;
      this.runNext();
    }
  }

  // 手动实现批量等待方法（不使用 Promise.all）
  async waitAll(promises) {
    const results = [];
    const errors = [];

    // 手动遍历所有 promise
    for (let i = 0; i < promises.length; i++) {
      try {
        const result = await promises[i];
        results[i] = result;
      } catch (error) {
        errors[i] = error;
      }
    }

    // 如果有错误，抛出第一个错误（模拟 Promise.all 的行为）
    if (errors.length > 0) {
      throw errors[0];
    }

    return results;
  }
}

// 使用示例
const pool = new RequestPool(3); // 最大并发数为3

// 模拟请求函数
function mockRequest(id, delay) {
  return new Promise(resolve => {
    setTimeout(() => {
      console.log(`Request ${id} completed`);
      resolve(`Result ${id}`);
    }, delay);
  });
}

// 添加多个请求到 pool（通过 pool.add 已经控制了并发）
const requests = [
  pool.add(() => mockRequest(1, 1000)),
  pool.add(() => mockRequest(2, 500)),
  pool.add(() => mockRequest(3, 800)),
  pool.add(() => mockRequest(4, 300)),
  pool.add(() => mockRequest(5, 600)),
  pool.add(() => mockRequest(6, 400)),
  pool.add(() => mockRequest(7, 700)),
];

// 使用 pool 的 waitAll 方法（不使用 Promise.all）
pool.waitAll(requests)
  .then(results => {
    console.log('All requests completed:', results);
  })
  .catch(error => {
    console.error('Error:', error);
  });
```

### 方案二：更简洁的实现方式

```javascript
class ConcurrencyController {
  constructor(maxConcurrent) {
    this.maxConcurrent = maxConcurrent;
    this.running = 0;
    this.queue = [];
  }

  async run(fn) {
    // 如果已达并发上限，加入队列等待
    while (this.running >= this.maxConcurrent) {
      await new Promise(resolve => this.queue.push(resolve));
    }

    this.running++;
    try {
      return await fn();
    } finally {
      this.running--;
      // 通知队列中的下一个任务
      if (this.queue.length > 0) {
        const next = this.queue.shift();
        next();
      }
    }
  }

  // 手动实现批量等待（不使用 Promise.all）
  async waitAll(promises) {
    const results = [];
    const errors = [];

    for (let i = 0; i < promises.length; i++) {
      try {
        const result = await promises[i];
        results[i] = result;
      } catch (error) {
        errors[i] = error;
      }
    }

    if (errors.length > 0) {
      throw errors[0];
    }

    return results;
  }
}

// 使用示例
const controller = new ConcurrencyController(3);

// 模拟请求函数
function mockRequest(id, delay) {
  return new Promise(resolve => {
    setTimeout(() => {
      console.log(`Request ${id} completed`);
      resolve(`Result ${id}`);
    }, delay);
  });
}

// 创建请求列表（通过 controller.run 包装，自动控制并发）
const requestList = [
  () => mockRequest(1, 1000),
  () => mockRequest(2, 500),
  () => mockRequest(3, 800),
  () => mockRequest(4, 300),
  () => mockRequest(5, 600),
  () => mockRequest(6, 400),
  () => mockRequest(7, 700),
];

// 将所有请求包装后执行
const promises = requestList.map(req => controller.run(req));

// 使用控制器的 waitAll 方法（不使用 Promise.all）
controller.waitAll(promises)
  .then(results => {
    console.log('All requests completed:', results);
  })
  .catch(error => {
    console.error('Error:', error);
  });
```

### 方案三：函数式实现

```javascript
// 创建并发控制器
function createConcurrencyLimit(limit) {
  let running = 0;
  const queue = [];

  return async function(fn) {
    // 如果已达并发上限，等待
    while (running >= limit) {
      await new Promise(resolve => queue.push(resolve));
    }

    running++;
    try {
      return await fn();
    } finally {
      running--;
      // 唤醒队列中的下一个
      if (queue.length > 0) {
        const next = queue.shift();
        next();
      }
    }
  };
}

// 手动实现 Promise.all 功能
async function manualPromiseAll(promises) {
  const results = [];
  const errors = [];

  for (let i = 0; i < promises.length; i++) {
    try {
      const result = await promises[i];
      results[i] = result;
    } catch (error) {
      errors[i] = error;
    }
  }

  if (errors.length > 0) {
    throw errors[0];
  }

  return results;
}

// 使用示例
const limit = createConcurrencyLimit(3);

// 模拟请求函数
function mockRequest(id, delay) {
  return new Promise(resolve => {
    setTimeout(() => {
      console.log(`Request ${id} completed`);
      resolve(`Result ${id}`);
    }, delay);
  });
}

// 创建请求列表
const tasks = [
  () => mockRequest(1, 1000),
  () => mockRequest(2, 500),
  () => mockRequest(3, 800),
  () => mockRequest(4, 300),
  () => mockRequest(5, 600),
  () => mockRequest(6, 400),
];

// 使用 limit 包装每个请求（控制并发）
const promises = tasks.map(task => limit(task));

// 使用 manualPromiseAll（不使用 Promise.all）
manualPromiseAll(promises)
  .then(results => {
    console.log('All requests completed:', results);
  })
  .catch(error => {
    console.error('Error:', error);
  });
```

---

### 核心思路说明：

1. **并发控制**：
   - 通过计数器 `running` 跟踪当前执行的任务数
   - 当达到上限时，新任务进入队列等待
   - 任务完成后，唤醒队列中的下一个任务

2. **手动实现 Promise.all**：
   - 使用 `for` 循环逐个 `await` 每个 promise
   - 按顺序收集结果到数组中
   - 如果有错误，抛出第一个错误（模拟 Promise.all 的行为）

3. **完全不使用 Promise.all**：
   - 所有并发控制都是手动实现
   - 所有结果收集都是手动遍历
   - 完全避免了使用 Promise.all 及其相关方法

这种方法真正实现了并发控制和批量等待，且完全不依赖 Promise.all。
