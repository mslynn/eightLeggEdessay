## 29. 两个字符串对比，得出结论都做了什么操作

**题目：** 比较两个字符串，得出做了什么操作（插入或删除）。

**参考答案：**

```javascript
function diffStrings(oldStr, newStr) {
  const m = oldStr.length;
  const n = newStr.length;
  const dp = Array(m + 1).fill(null).map(() => Array(n + 1).fill(0));

  // 构建DP矩阵
  for (let i = 0; i <= m; i++) {
    for (let j = 0; j <= n; j++) {
      if (i === 0) {
        dp[i][j] = j; // 需要j次插入
      } else if (j === 0) {
        dp[i][j] = i; // 需要i次删除
      } else if (oldStr[i - 1] === newStr[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1];
      } else {
        dp[i][j] = 1 + Math.min(
          dp[i - 1][j],    // 删除
          dp[i][j - 1],    // 插入
          dp[i - 1][j - 1] // 替换
        );
      }
    }
  }

  // 回溯找出操作序列
  const operations = [];
  let i = m, j = n;

  while (i > 0 || j > 0) {
    if (i > 0 && j > 0 && oldStr[i - 1] === newStr[j - 1]) {
      i--;
      j--;
    } else if (j > 0 && (i === 0 || dp[i][j - 1] <= dp[i - 1][j])) {
      operations.unshift({
        type: 'insert',
        char: newStr[j - 1],
        position: i
      });
      j--;
    } else if (i > 0) {
      operations.unshift({
        type: 'delete',
        char: oldStr[i - 1],
        position: i - 1
      });
      i--;
    }
  }

  return operations;
}

// 测试
const oldStr = 'abcde';
const newStr = 'abXdeY';
const diff = diffStrings(oldStr, newStr);
console.log('差异操作:', diff);
// [
//   { type: 'delete', char: 'c', position: 2 },
//   { type: 'insert', char: 'X', position: 2 },
//   { type: 'delete', char: 'd', position: 3 },
//   { type: 'insert', char: 'd', position: 3 },
//   { type: 'insert', char: 'Y', position: 4 }
// ]

// 更简单的版本（只统计插入和删除）
function diffStringsSimple(oldStr, newStr) {
  const oldChars = oldStr.split('');
  const newChars = newStr.split('');
  const inserted = [];
  const deleted = [];

  // 找出插入的字符
  for (const char of newChars) {
    const index = oldChars.indexOf(char);
    if (index === -1) {
      inserted.push(char);
    } else {
      oldChars.splice(index, 1);
    }
  }

  // 剩下的就是删除的字符
  deleted.push(...oldChars);

  return { inserted, deleted };
}

console.log('简单差异:', diffStringsSimple('abcde', 'abXdeY'));
// { inserted: ['X', 'Y'], deleted: ['c'] }
```
