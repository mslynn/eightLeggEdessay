## 35. 使用TS实现一个判断传入参数是否是数组类型的方法

**题目：** 使用TypeScript实现判断参数是否为数组类型的方法。

**参考答案：**

```typescript
// 方法1：使用Array.isArray
function isArray1(value: unknown): value is Array<unknown> {
  return Array.isArray(value);
}

// 方法2：使用类型守卫
function isArray2<T>(value: unknown): value is T[] {
  return Array.isArray(value);
}

// 方法3：使用instanceof（不推荐，跨iframe有问题）
function isArray3(value: unknown): value is any[] {
  return value instanceof Array;
}

// 方法4：使用Object.prototype.toString
function isArray4(value: unknown): value is any[] {
  return Object.prototype.toString.call(value) === '[object Array]';
}

// 测试
const test1: unknown = [1, 2, 3];
const test2: unknown = 'hello';
const test3: unknown = { a: 1 };

console.log(isArray1(test1)); // true
console.log(isArray1(test2)); // false
console.log(isArray1(test3)); // false

if (isArray2<number>(test1)) {
  console.log(test1.map(x => x * 2)); // [2, 4, 6]
}

if (isArray1(test2)) {
  console.log(test2.length); // 不会执行
}

// 泛型版本
function isArrayOfType<T>(value: unknown, typeGuard: (item: unknown) => item is T): value is T[] {
  if (!Array.isArray(value)) {
    return false;
  }
  return value.every(typeGuard);
}

// 使用示例
function isNumber(value: unknown): value is number {
  return typeof value === 'number';
}

const test4: unknown = [1, 2, 3];
const test5: unknown = [1, 'hello', 3];

if (isArrayOfType(test4, isNumber)) {
  console.log(test4.reduce((a, b) => a + b, 0)); // 6
}

if (isArrayOfType(test5, isNumber)) {
  console.log('All numbers');
} else {
  console.log('Not all numbers');
}
```
