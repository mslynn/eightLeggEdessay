## 34. 实现一个将多维数组扁平化的方法

**题目：** 实现数组扁平化，将多维数组转换为一维数组。

**参考答案：**

```javascript
// 方法1：使用flat（ES6）
function flatten1(arr, depth = Infinity) {
  return arr.flat(depth);
}

// 方法2：递归
function flatten2(arr) {
  const result = [];

  for (const item of arr) {
    if (Array.isArray(item)) {
      result.push(...flatten2(item));
    } else {
      result.push(item);
    }
  }

  return result;
}

// 方法3：使用reduce
function flatten3(arr) {
  return arr.reduce((acc, item) => {
    return acc.concat(Array.isArray(item) ? flatten3(item) : item);
  }, []);
}

// 方法4：使用toString（只适用于数字）
function flatten4(arr) {
  return arr.toString().split(',').map(Number);
}

// 方法5：使用堆栈
function flatten5(arr) {
  const stack = [...arr];
  const result = [];

  while (stack.length) {
    const item = stack.pop();

    if (Array.isArray(item)) {
      stack.push(...item);
    } else {
      result.push(item);
    }
  }

  return result.reverse();
}

// 测试
const arr = [1, [2, [3, [4, 5]]], 6, [7, 8]];

console.log(flatten1(arr)); // [1, 2, 3, 4, 5, 6, 7, 8]
console.log(flatten2(arr)); // [1, 2, 3, 4, 5, 6, 7, 8]
console.log(flatten3(arr)); // [1, 2, 3, 4, 5, 6, 7, 8]
console.log(flatten4(arr)); // [1, 2, 3, 4, 5, 6, 7, 8]
console.log(flatten5(arr)); // [1, 2, 3, 4, 5, 6, 7, 8]

// 指定扁平化深度
const arr2 = [1, [2, [3, [4, 5]]], 6];
console.log(flatten1(arr2, 1)); // [1, 2, [3, [4, 5]], 6]
console.log(flatten1(arr2, 2)); // [1, 2, 3, [4, 5], 6]
```
