## 32. 使用正则表达式实现以下需求：筛选出数组中只包含大小写字母的字符串，并将结果转换成大写

**题目：** 使用正则表达式筛选出只包含大小写字母的字符串，并转换成大写。

**参考答案：**

```javascript
function filterAndConvertToUpper(arr) {
  return arr
    .filter(str => /^[a-zA-Z]+$/.test(str))
    .map(str => str.toUpperCase());
}

// 测试
const arr = ['hello', 'world123', 'ABC', 'aBc', 'test!', 'JavaScript', '123', ''];
console.log(filterAndConvertToUpper(arr));
// ['HELLO', 'ABC', 'ABC', 'JAVASCRIPT']

// 更详细的版本
function filterAndConvertToUpperDetailed(arr) {
  return arr
    .filter(str => {
      // 只包含字母
      const isOnlyLetters = /^[a-zA-Z]+$/.test(str);
      console.log(`"${str}" 只包含字母: ${isOnlyLetters}`);
      return isOnlyLetters;
    })
    .map(str => {
      const upperStr = str.toUpperCase();
      console.log(`"${str}" -> "${upperStr}"`);
      return upperStr;
    });
}

// filterAndConvertToUpperDetailed(arr);

// 使用正则表达式的另一种方式
function filterAndConvertToUpper2(arr) {
  return arr.reduce((result, str) => {
    if (/^[a-zA-Z]+$/.test(str)) {
      result.push(str.toUpperCase());
    }
    return result;
  }, []);
}

console.log(filterAndConvertToUpper2(arr));
// ['HELLO', 'ABC', 'ABC', 'JAVASCRIPT']
```
