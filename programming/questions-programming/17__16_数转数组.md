## 16. 树转数组

**题目：** 将树状结构转换为数组，实现扁平化操作。

**参考答案：**

```javascript
// 方法1：深度优先遍历（DFS）- 递归
function treeToArrayDFS(tree) {
  const result = [];

  function traverse(node) {
    if (!node) return;

    result.push(node.value);

    if (node.children && node.children.length > 0) {
      node.children.forEach(child => traverse(child));
    }
  }

  traverse(tree);
  return result;
}

// 方法2：深度优先遍历（DFS）- 迭代（栈）
function treeToArrayDFSIterative(tree) {
  const result = [];
  const stack = [tree];

  while (stack.length > 0) {
    const node = stack.pop();
    result.push(node.value);

    if (node.children && node.children.length > 0) {
      // 反向入栈，保证从左到右遍历
      for (let i = node.children.length - 1; i >= 0; i--) {
        stack.push(node.children[i]);
      }
    }
  }

  return result;
}

// 方法3：广度优先遍历（BFS）- 队列
function treeToArrayBFS(tree) {
  const result = [];
  const queue = [tree];

  while (queue.length > 0) {
    const node = queue.shift();
    result.push(node.value);

    if (node.children && node.children.length > 0) {
      node.children.forEach(child => queue.push(child));
    }
  }

  return result;
}

// 方法4：返回完整路径的数组
function treeToPathArray(tree, path = []) {
  const result = [];

  function traverse(node, currentPath) {
    if (!node) return;

    const newPath = [...currentPath, node.value];
    result.push(newPath);

    if (node.children && node.children.length > 0) {
      node.children.forEach(child => traverse(child, newPath));
    }
  }

  traverse(tree, path);
  return result;
}

// 测试数据
const tree = {
  value: 1,
  children: [
    {
      value: 2,
      children: [
        { value: 4, children: [] },
        { value: 5, children: [] }
      ]
    },
    {
      value: 3,
      children: [
        { value: 6, children: [] },
        { value: 7, children: [] }
      ]
    }
  ]
};

console.log(treeToArrayDFS(tree));           // [1, 2, 4, 5, 3, 6, 7]
console.log(treeToArrayDFSIterative(tree));  // [1, 2, 4, 5, 3, 6, 7]
console.log(treeToArrayBFS(tree));           // [1, 2, 3, 4, 5, 6, 7]
console.log(treeToPathArray(tree));          // [[1], [1, 2], [1, 2, 4], [1, 2, 5], [1, 3], [1, 3, 6], [1, 3, 7]]
```
