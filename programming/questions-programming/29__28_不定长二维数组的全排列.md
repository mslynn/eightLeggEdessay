## 28. 不定长二维数组的全排列

**题目：** 给定不定长二维数组，输出所有可能的组合。如 `[['A','B'], ['a','b'], [1, 2]]` 输出 `['Aa1','Aa2','Ab1','Ab2','Ba1','Ba2','Bb1','Bb2']`

**参考答案：**

```javascript
function cartesianProduct(arrays) {
  if (arrays.length === 0) {
    return [];
  }

  let result = arrays[0].map(item => [item]);

  for (let i = 1; i < arrays.length; i++) {
    const temp = [];
    for (const item1 of result) {
      for (const item2 of arrays[i]) {
        temp.push([...item1, item2]);
      }
    }
    result = temp;
  }

  return result.map(item => item.join(''));
}

// 方法2：使用递归
function cartesianProductRecursive(arrays, index = 0) {
  if (index === arrays.length) {
    return [''];
  }

  const current = arrays[index];
  const rest = cartesianProductRecursive(arrays, index + 1);

  const result = [];
  for (const item of current) {
    for (const item2 of rest) {
      result.push(item + item2);
    }
  }

  return result;
}

// 方法3：使用reduce
function cartesianProductReduce(arrays) {
  return arrays.reduce((acc, curr) => {
    return acc.flatMap(item => curr.map(c => item + c));
  }, ['']);
}

// 测试
const arrays = [['A', 'B'], ['a', 'b'], [1, 2]];
console.log('笛卡尔积:', cartesianProduct(arrays));
// ['Aa1', 'Aa2', 'Ab1', 'Ab2', 'Ba1', 'Ba2', 'Bb1', 'Bb2']

console.log('递归版本:', cartesianProductRecursive(arrays));
// ['Aa1', 'Aa2', 'Ab1', 'Ab2', 'Ba1', 'Ba2', 'Bb1', 'Bb2']

console.log('Reduce版本:', cartesianProductReduce(arrays));
// ['Aa1', 'Aa2', 'Ab1', 'Ab2', 'Ba1', 'Ba2', 'Bb1', 'Bb2']

// 不定长数组测试
const arrays2 = [['A', 'B', 'C'], ['a'], [1, 2, 3]];
console.log(cartesianProduct(arrays2));
// ['Aa1', 'Aa2', 'Aa3', 'Ba1', 'Ba2', 'Ba3', 'Ca1', 'Ca2', 'Ca3']
```
