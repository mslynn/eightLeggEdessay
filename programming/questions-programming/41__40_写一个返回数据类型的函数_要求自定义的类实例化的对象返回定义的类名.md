## 40. 写一个返回数据类型的函数，要求自定义的类实例化的对象返回定义的类名

**题目：** 实现一个类型判断函数，对于自定义类实例返回类名。

**参考答案：**

```javascript
function myTypeof(data) {
  const toString = Object.prototype.toString;
  const dataType = data instanceof Element
    ? "Element"
    : toString.call(data).replace(/\[object (\w+)\]/, "$1");

  if (dataType === 'Object') {
    return data.constructor.name;
  }

  return dataType;
}

// 测试
console.log(myTypeof("hello"));         // String
console.log(myTypeof(123));              // Number
console.log(myTypeof(true));             // Boolean
console.log(myTypeof(null));             // Null
console.log(myTypeof(undefined));        // Undefined
console.log(myTypeof([1, 2, 3]));        // Array
console.log(myTypeof({ a: 1 }));         // Object
console.log(myTypeof(new Date()));        // Date
console.log(myTypeof(/test/));           // RegExp
console.log(myTypeof(function() {}));    // Function

// 自定义类
class Person {
  constructor(name) {
    this.name = name;
  }
}

class Animal {
  constructor(type) {
    this.type = type;
  }
}

console.log(myTypeof(new Person('Alice')));  // Person
console.log(myTypeof(new Animal('Dog')));    // Animal

// DOM元素
console.log(myTypeof(document.body));        // Element
console.log(myTypeof(document.querySelector('div'))); // Element

// 更完善的版本
function myTypeofAdvanced(data) {
  // 处理null和undefined
  if (data === null) return 'Null';
  if (data === undefined) return 'Undefined';

  // 处理基本类型
  const type = typeof data;
  if (type !== 'object' && type !== 'function') {
    return type.charAt(0).toUpperCase() + type.slice(1);
  }

  // 处理Element
  if (data instanceof Element) {
    return 'Element';
  }

  // 处理其他对象类型
  const toString = Object.prototype.toString.call(data);
  const match = toString.match(/\[object (\w+)\]/);

  if (match) {
    const objType = match[1];

    // 如果是Object，返回构造函数名
    if (objType === 'Object' && data.constructor && data.constructor.name) {
      return data.constructor.name;
    }

    return objType;
  }

  return 'Unknown';
}

console.log(myTypeofAdvanced(new Person('Alice')));  // Person
console.log(myTypeofAdvanced(new Map()));            // Map
console.log(myTypeofAdvanced(new Set()));            // Set
console.log(myTypeofAdvanced(new Promise(() => {}))); // Promise
```
