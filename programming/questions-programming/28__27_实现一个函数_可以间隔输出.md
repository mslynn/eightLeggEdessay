## 27. 实现一个函数，可以间隔输出

**题目：** 实现一个createRepeat函数，可以间隔输出指定次数。

**参考答案：**

```javascript
function createRepeat(fn, times, interval) {
  let count = 0;
  let timer = null;

  return function(...args) {
    if (timer) {
      clearInterval(timer);
    }

    timer = setInterval(() => {
      if (count >= times) {
        clearInterval(timer);
        timer = null;
        return;
      }

      fn.apply(this, args);
      count++;
    }, interval);
  };
}

// 测试
const repeatLog = createRepeat(console.log, 3, 1000);
repeatLog('Hello World');

// 输出：
// Hello World (1秒后)
// Hello World (2秒后)
// Hello World (3秒后)

// 更灵活的版本
function createRepeatAdvanced(fn, times, interval, immediate = false) {
  let count = 0;
  let timer = null;

  return function(...args) {
    if (timer) {
      clearInterval(timer);
    }

    if (immediate) {
      fn.apply(this, args);
      count++;
    }

    timer = setInterval(() => {
      if (count >= times) {
        clearInterval(timer);
        timer = null;
        return;
      }

      fn.apply(this, args);
      count++;
    }, interval);

    // 返回取消函数
    return () => {
      if (timer) {
        clearInterval(timer);
        timer = null;
      }
    };
  };
}

// 测试
const repeatLog2 = createRepeatAdvanced(console.log, 3, 1000, true);
const cancel = repeatLog2('Hello World');

// 输出：
// Hello World (立即)
// Hello World (1秒后)
// Hello World (2秒后)
// Hello World (3秒后)

// 取消
// cancel();
```
