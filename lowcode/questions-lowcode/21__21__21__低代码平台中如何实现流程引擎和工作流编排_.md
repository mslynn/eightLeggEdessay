## 21. 低代码平台中如何实现流程引擎和工作流编排?

**答案:**

流程引擎是低代码平台处理业务流程自动化的核心组件,支持可视化流程设计、流程执行和流程监控。

### 1. 流程引擎核心概念

### 2. 流程可视化编辑器

```javascript
class ProcessDesigner {
  constructor(container) {
    this.container = container;
    this.canvas = null;
    this.nodes = [];
    this.edges = [];
    this.selectedNode = null;
    this.draggingNode = null;
    this.dragOffset = { x: 0, y: 0 };
    
    this.init();
  }
  
  // 初始化画布
  init() {
    this.canvas = document.createElement('div');
    this.canvas.className = 'process-canvas';
    this.canvas.style.cssText = `
      position: relative;
      width: 100%;
      height: 100%;
      background: #f5f5f5;
      overflow: auto;
    `;
    
    this.container.appendChild(this.canvas);
    this.setupEventListeners();
  }
  
  // 添加节点
  addNode(nodeConfig) {
    const node = {
      ...nodeConfig,
      id: `node-${Date.now()}`
    };
    
    this.nodes.push(node);
    this.renderNode(node);
    return node;
  }
  
  // 渲染节点
  renderNode(node) {
    const element = document.createElement('div');
    element.className = `process-node ${node.type}`;
    element.dataset.nodeId = node.id;
    element.style.cssText = `
      position: absolute;
      left: ${node.position.x}px;
      top: ${node.position.y}px;
      width: 120px;
      height: 60px;
      background: white;
      border: 2px solid #1890ff;
      border-radius: 4px;
      display: flex;
      align-items: center;
      justify-content: center;
      cursor: move;
      user-select: none;
      z-index: 1;
    `;
    
    element.innerHTML = `
      <span style="font-size: 12px;">${node.name}</span>
    `;
    
    this.canvas.appendChild(element);
    
    // 添加拖拽事件
    element.addEventListener('mousedown', (e) => this.handleNodeMouseDown(e, node));
  }
  
  // 添加连接线
  addEdge(edgeConfig) {
    const edge = {
      ...edgeConfig,
      id: `edge-${Date.now()}`
    };
    
    this.edges.push(edge);
    this.renderEdge(edge);
    return edge;
  }
  
  // 渲染连接线
  renderEdge(edge) {
    const sourceNode = this.nodes.find(n => n.id === edge.source);
    const targetNode = this.nodes.find(n => n.id === edge.target);
    
    if (!sourceNode || !targetNode) return;
    
    const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
    svg.style.cssText = `
      position: absolute;
      left: 0;
      top: 0;
      width: 100%;
      height: 100%;
      pointer-events: none;
      z-index: 0;
    `;
    
    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    
    const sourceX = sourceNode.position.x + 60;
    const sourceY = sourceNode.position.y + 30;
    const targetX = targetNode.position.x;
    const targetY = targetNode.position.y + 30;
    
    // 贝塞尔曲线
    const controlX1 = sourceX + 50;
    const controlX2 = targetX - 50;
    
    path.setAttribute('d', `M ${sourceX} ${sourceY} C ${controlX1} ${sourceY}, ${controlX2} ${targetY}, ${targetX} ${targetY}`);
    path.setAttribute('stroke', '#1890ff');
    path.setAttribute('stroke-width', '2');
    path.setAttribute('fill', 'none');
    path.setAttribute('marker-end', 'url(#arrow)');
    
    svg.appendChild(path);
    this.canvas.appendChild(svg);
  }
  
  // 节点拖拽处理
  handleNodeMouseDown(e, node) {
    e.stopPropagation();
    this.draggingNode = node;
    this.dragOffset = {
      x: e.clientX - node.position.x,
      y: e.clientY - node.position.y
    };
  }
  
  // 画布事件监听
  setupEventListeners() {
    this.canvas.addEventListener('mousemove', (e) => {
      if (this.draggingNode) {
        const rect = this.canvas.getBoundingClientRect();
        this.draggingNode.position.x = e.clientX - rect.left - this.dragOffset.x;
        this.draggingNode.position.y = e.clientY - rect.top - this.dragOffset.y;
        
        // 更新节点位置
        const element = this.canvas.querySelector(`[data-node-id="${this.draggingNode.id}"]`);
        if (element) {
          element.style.left = `${this.draggingNode.position.x}px`;
          element.style.top = `${this.draggingNode.position.y}px`;
        }
        
        // 重新渲染连接线
        this.renderEdges();
      }
    });
    
    this.canvas.addEventListener('mouseup', () => {
      this.draggingNode = null;
    });
  }
  
  // 渲染所有连接线
  renderEdges() {
    // 移除旧的SVG
    const svgs = this.canvas.querySelectorAll('svg');
    svgs.forEach(svg => svg.remove());
    
    // 重新渲染
    this.edges.forEach(edge => this.renderEdge(edge));
  }
}
```

### 3. 流程运行引擎

```javascript
class ProcessEngine {
  constructor() {
    this.processInstances = new Map();
    this.taskListeners = [];
  }
  
  // 启动流程实例
  startProcess(processDefinition, variables = {}) {
    const instance = {
      instanceId: `instance-${Date.now()}`,
      processId: processDefinition.processId,
      startTime: new Date(),
      variables: variables,
      currentNodeId: null,
      status: 'running',
      history: []
    };
    
    // 找到开始节点
    const startNode = processDefinition.nodes.find(n => n.type === 'start');
    if (startNode) {
      instance.currentNodeId = startNode.id;
      instance.history.push({
        nodeId: startNode.id,
        time: new Date()
      });
    }
    
    this.processInstances.set(instance.instanceId, instance);
    this.executeNextStep(instance, processDefinition);
    
    return instance;
  }
  
  // 执行下一步
  executeNextStep(instance, processDefinition) {
    const currentNode = processDefinition.nodes.find(n => n.id === instance.currentNodeId);
    
    if (!currentNode) return;
    
    switch (currentNode.type) {
      case 'start':
        // 开始节点自动流转到下一个节点
        this.findAndExecuteNextNode(instance, currentNode, processDefinition);
        break;
        
      case 'userTask':
        // 用户任务节点,等待用户操作
        this.createTask(instance, currentNode);
        break;
        
      case 'exclusiveGateway':
        // 排他网关,根据条件流转
        this.executeGateway(instance, currentNode, processDefinition);
        break;
        
      case 'end':
        // 结束节点
        instance.status = 'completed';
        instance.endTime = new Date();
        break;
    }
  }
  
  // 创建用户任务
  createTask(instance, node) {
    const task = {
      taskId: `task-${Date.now()}`,
      instanceId: instance.instanceId,
      nodeId: node.id,
      name: node.name,
      assignee: this.evaluateExpression(node.assignee, instance.variables),
      createTime: new Date(),
      status: 'pending',
      properties: node.properties || {}
    };
    
    // 通知任务监听器
    this.taskListeners.forEach(listener => {
      listener.onTaskCreated(task);
    });
  }
  
  // 完成任务
  completeTask(taskId, variables) {
    // 查找任务
    const task = this.findTask(taskId);
    if (!task) {
      throw new Error(`任务 ${taskId} 不存在`);
    }
    
    const instance = this.processInstances.get(task.instanceId);
    if (!instance) {
      throw new Error(`流程实例 ${task.instanceId} 不存在`);
    }
    
    // 更新变量
    Object.assign(instance.variables, variables);
    
    // 更新任务状态
    task.status = 'completed';
    task.completeTime = new Date();
    
    // 记录历史
    instance.history.push({
      nodeId: task.nodeId,
      time: new Date(),
      type: 'task-complete'
    });
    
    // 执行下一步
    const processDefinition = this.getProcessDefinition(instance.processId);
    const currentNode = processDefinition.nodes.find(n => n.id === task.nodeId);
    this.findAndExecuteNextNode(instance, currentNode, processDefinition);
  }
  
  // 执行网关
  executeGateway(instance, gatewayNode, processDefinition) {
    // 查找所有从网关出发的连接线
    const outgoingEdges = processDefinition.edges.filter(e => e.source === gatewayNode.id);
    
    // 评估每条连接线的条件
    for (const edge of outgoingEdges) {
      if (edge.condition) {
        const conditionResult = this.evaluateExpression(edge.condition, instance.variables);
        if (conditionResult) {
          // 条件满足,流转到目标节点
          instance.currentNodeId = edge.target;
          instance.history.push({
            nodeId: edge.target,
            time: new Date(),
            type: 'gateway-transition'
          });
          this.executeNextStep(instance, processDefinition);
          return;
        }
      }
    }
    
    // 没有满足的条件,流程结束
    instance.status = 'terminated';
  }
  
  // 查找并执行下一个节点
  findAndExecuteNextNode(instance, currentNode, processDefinition) {
    const outgoingEdge = processDefinition.edges.find(e => e.source === currentNode.id);
    
    if (outgoingEdge) {
      instance.currentNodeId = outgoingEdge.target;
      instance.history.push({
        nodeId: outgoingEdge.target,
        time: new Date()
      });
      this.executeNextStep(instance, processDefinition);
    }
  }
  
  // 表达式求值
  evaluateExpression(expression, variables) {
    // 简化实现,实际可以使用更强大的表达式引擎
    const func = new Function('variables', `with(variables) { return ${expression}; }`);
    return func(variables);
  }
  
  // 添加任务监听器
  addTaskListener(listener) {
    this.taskListeners.push(listener);
  }
}
```

### 4. 流程监控

```javascript
class ProcessMonitor {
  constructor(processEngine) {
    this.processEngine = processEngine;
  }
  
  // 获取流程实例状态
  getInstanceStatus(instanceId) {
    const instance = this.processEngine.processInstances.get(instanceId);
    if (!instance) {
      return null;
    }
    
    return {
      instanceId: instance.instanceId,
      status: instance.status,
      currentNodeId: instance.currentNodeId,
      variables: instance.variables,
      history: instance.history,
      duration: instance.endTime
        ? instance.endTime - instance.startTime
        : Date.now() - instance.startTime
    };
  }
  
  // 获取待办任务列表
  getPendingTasks(userId) {
    const tasks = [];
    
    for (const instance of this.processEngine.processInstances.values()) {
      if (instance.status === 'running') {
        const task = this.findTaskByInstance(instance.instanceId, userId);
        if (task && task.status === 'pending') {
          tasks.push(task);
        }
      }
    }
    
    return tasks;
  }
  
  // 流程统计
  getProcessStatistics(processId) {
    const instances = Array.from(this.processEngine.processInstances.values())
      .filter(i => i.processId === processId);
    
    return {
      total: instances.length,
      running: instances.filter(i => i.status === 'running').length,
      completed: instances.filter(i => i.status === 'completed').length,
      terminated: instances.filter(i => i.status === 'terminated').length,
      avgDuration: instances
        .filter(i => i.endTime)
        .reduce((sum, i) => sum + (i.endTime - i.startTime), 0) / instances.length
    };
  }
}
```
