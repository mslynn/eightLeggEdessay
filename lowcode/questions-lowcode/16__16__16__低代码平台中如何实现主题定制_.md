## 16. 低代码平台中如何实现主题定制?

**答案:**

主题定制是低代码平台提供个性化外观的重要功能,支持自定义颜色、字体、间距等样式。

### 2. 主题管理器

```javascript
class ThemeManager {
  constructor() {
    this.currentTheme = 'default';
    this.themes = new Map();
    this.customProperties = new Map();
    this.listeners = [];
  }
  
  // 注册主题
  registerTheme(name, theme) {
    this.themes.set(name, theme);
  }
  
  // 设置当前主题
  setTheme(name) {
    if (!this.themes.has(name)) {
      console.warn(`主题 ${name} 不存在`);
      return;
    }
    
    this.currentTheme = name;
    this.applyTheme();
    this.notifyListeners();
  }
  
  // 获取当前主题
  getTheme() {
    return this.themes.get(this.currentTheme);
  }
  
  // 获取主题值
  getValue(path, defaultValue) {
    const theme = this.getTheme();
    const keys = path.split('.');
    let value = theme;
    
    for (const key of keys) {
      if (value && typeof value === 'object') {
        value = value[key];
      } else {
        value = defaultValue;
        break;
      }
    }
    
    return value !== undefined ? value : defaultValue;
  }
  
  // 应用主题到CSS变量
  applyTheme() {
    const theme = this.getTheme();
    const root = document.documentElement;
    
    // 颜色
    root.style.setProperty('--primary-color', theme.primaryColor);
    root.style.setProperty('--success-color', theme.successColor);
    root.style.setProperty('--warning-color', theme.warningColor);
    root.style.setProperty('--error-color', theme.errorColor);
    root.style.setProperty('--info-color', theme.infoColor);
    root.style.setProperty('--text-color', theme.textColor);
    root.style.setProperty('--text-secondary-color', theme.textSecondaryColor);
    root.style.setProperty('--disabled-color', theme.disabledColor);
    root.style.setProperty('--border-color', theme.borderColor);
    root.style.setProperty('--background-color', theme.backgroundColor);
    root.style.setProperty('--page-background-color', theme.pageBackgroundColor);
    
    // 字体
    root.style.setProperty('--font-family', theme.fontFamily);
    root.style.setProperty('--font-size', `${theme.fontSize}px`);
    root.style.setProperty('--font-size-small', `${theme.fontSizeSmall}px`);
    root.style.setProperty('--font-size-large', `${theme.fontSizeLarge}px`);
    root.style.setProperty('--font-size-x-large', `${theme.fontSizeXLarge}px`);
    
    // 间距
    root.style.setProperty('--spacing-xs', `${theme.spacing.xs}px`);
    root.style.setProperty('--spacing-sm', `${theme.spacing.sm}px`);
    root.style.setProperty('--spacing-md', `${theme.spacing.md}px`);
    root.style.setProperty('--spacing-lg', `${theme.spacing.lg}px`);
    root.style.setProperty('--spacing-xl', `${theme.spacing.xl}px`);
    
    // 圆角
    root.style.setProperty('--border-radius', `${theme.borderRadius}px`);
    root.style.setProperty('--border-radius-large', `${theme.borderRadiusLarge}px`);
    
    // 阴影
    root.style.setProperty('--box-shadow', theme.boxShadow);
    root.style.setProperty('--box-shadow-large', theme.boxShadowLarge);
  }
  
  // 自定义主题属性
  setCustomProperty(name, value) {
    this.customProperties.set(name, value);
    document.documentElement.style.setProperty(name, value);
  }
  
  // 添加主题变化监听器
  addListener(listener) {
    this.listeners.push(listener);
    return () => {
      const index = this.listeners.indexOf(listener);
      if (index !== -1) {
        this.listeners.splice(index, 1);
      }
    };
  }
  
  // 通知监听器
  notifyListeners() {
    this.listeners.forEach(listener => listener(this.currentTheme));
  }
}
```

### 4. React集成

```javascript
// 创建主题上下文
const ThemeContext = React.createContext();

// 主题提供者组件
export const ThemeProvider = ({ children, themeManager }) => {
  const [theme, setTheme] = React.useState(themeManager.getTheme());
  
  React.useEffect(() => {
    const unsubscribe = themeManager.addListener((themeName) => {
      setTheme(themeManager.getTheme());
    });
    
    return unsubscribe;
  }, [themeManager]);
  
  const value = {
    theme,
    themeName: themeManager.currentTheme,
    setTheme: (name) => themeManager.setTheme(name),
    getValue: (path, defaultValue) => themeManager.getValue(path, defaultValue)
  };
  
  return (
    <ThemeContext.Provider value={value}>
      {children}
    </ThemeContext.Provider>
  );
};

// 使用主题的Hook
export const useTheme = () => {
  const context = React.useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider');
  }
  return context;
};

// 主题按钮组件
export const ThemeButton = () => {
  const { themeName, setTheme } = useTheme();
  
  const themes = [
    { name: 'default', label: '蓝色' },
    { name: 'dark', label: '暗色' },
    { name: 'green', label: '绿色' },
    { name: 'purple', label: '紫色' }
  ];
  
  return (
    <select value={themeName} onChange={(e) => setTheme(e.target.value)}>
      {themes.map(t => (
        <option key={t.name} value={t.name}>{t.label}</option>
      ))}
    </select>
  );
};
```
