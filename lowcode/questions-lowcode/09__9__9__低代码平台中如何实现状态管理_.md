## 9. 低代码平台中如何实现状态管理?

**答案:**

低代码平台需要一套强大的状态管理系统来处理组件间的数据共享和状态同步。

### 1. 全局状态设计

```javascript
class GlobalStore {
  constructor() {
    this.state = {};
    this.listeners = [];
    this.mutations = {};
    this.actions = {};
  }
  
  // 获取状态
  getState(key) {
    return key ? this.state[key] : this.state;
  }
  
  // 设置状态
  setState(key, value) {
    this.state[key] = value;
    this.notify();
  }
  
  // 注册变更监听
  subscribe(listener) {
    this.listeners.push(listener);
    return () => {
      const index = this.listeners.indexOf(listener);
      this.listeners.splice(index, 1);
    };
  }
  
  // 通知变更
  notify() {
    this.listeners.forEach(listener => listener(this.state));
  }
  
  // 注册mutation
  registerMutation(name, handler) {
    this.mutations[name] = handler;
  }
  
  // 执行mutation
  commit(name, payload) {
    if (this.mutations[name]) {
      this.mutations[name](this.state, payload);
      this.notify();
    }
  }
  
  // 注册action
  registerAction(name, handler) {
    this.actions[name] = handler;
  }
  
  // 执行action
  async dispatch(name, payload) {
    if (this.actions[name]) {
      await this.actions[name]({ state: this.state, commit: this.commit.bind(this) }, payload);
    }
  }
}
```

### 2. 状态初始化

```javascript
const store = new GlobalStore();

// 定义初始状态
store.setState('user', {
  name: '',
  avatar: '',
  role: 'guest'
});

store.setState('app', {
  theme: 'light',
  language: 'zh-CN',
  loading: false
});

// 定义mutations
store.registerMutation('setUser', (state, user) => {
  state.user = user;
});

store.registerMutation('setTheme', (state, theme) => {
  state.app.theme = theme;
});

// 定义actions
store.registerAction('fetchUserInfo', async ({ commit }, userId) => {
  const user = await api.getUser(userId);
  commit('setUser', user);
});
```

### 4. 页面级状态

```javascript
class PageStore extends GlobalStore {
  constructor(pageId) {
    super();
    this.pageId = pageId;
    this.state = {
      formData: {},
      loading: false,
      error: null
    };
  }
  
  // 表单数据更新
  updateFormData(field, value) {
    this.state.formData[field] = value;
    this.notify();
  }
  
  // 提交表单
  async submitForm(apiUrl) {
    this.state.loading = true;
    this.notify();
    
    try {
      const result = await api.post(apiUrl, this.state.formData);
      this.state.loading = false;
      this.notify();
      return result;
    } catch (error) {
      this.state.error = error;
      this.state.loading = false;
      this.notify();
      throw error;
    }
  }
}
```

### 5. 数据绑定实现

```javascript
function createBinding(expression, context) {
  const parts = expression.match(/\$\{([^}]+)\}/);
  
  if (!parts) {
    return expression;
  }
  
  const path = parts[1];
  
  // 监听状态变化
  const unsubscribe = context.store.subscribe((state) => {
    const value = getValueByPath(state, path);
    updateElement(expression, value);
  });
  
  // 获取初始值
  const initialValue = getValueByPath(context.store.getState(), path);
  return initialValue;
}

function getValueByPath(obj, path) {
  return path.split('.').reduce((acc, key) => acc?.[key], obj);
}
```

### 6. 状态持久化

```javascript
class PersistStore extends GlobalStore {
  constructor(key, storage = localStorage) {
    super();
    this.key = key;
    this.storage = storage;
    this.loadState();
  }
  
  // 加载持久化状态
  loadState() {
    const saved = this.storage.getItem(this.key);
    if (saved) {
      try {
        this.state = JSON.parse(saved);
      } catch (e) {
        console.error('加载状态失败:', e);
      }
    }
  }
  
  // 保存状态
  saveState() {
    this.storage.setItem(this.key, JSON.stringify(this.state));
  }
  
  // 重写notify方法
  notify() {
    super.notify();
    this.saveState();
  }
}
```

### 7. 状态回滚

```javascript
class HistoryStore extends GlobalStore {
  constructor() {
    super();
    this.history = [];
    this.currentIndex = -1;
    this.maxHistory = 50;
  }
  
  // 记录状态快照
  recordSnapshot() {
    const snapshot = JSON.parse(JSON.stringify(this.state));
    
    // 删除当前位置之后的历史
    this.history = this.history.slice(0, this.currentIndex + 1);
    
    // 添加新快照
    this.history.push(snapshot);
    
    // 限制历史长度
    if (this.history.length > this.maxHistory) {
      this.history.shift();
    } else {
      this.currentIndex++;
    }
  }
  
  // 撤销
  undo() {
    if (this.currentIndex > 0) {
      this.currentIndex--;
      this.state = JSON.parse(JSON.stringify(this.history[this.currentIndex]));
      this.notify();
    }
  }
  
  // 重做
  redo() {
    if (this.currentIndex < this.history.length - 1) {
      this.currentIndex++;
      this.state = JSON.parse(JSON.stringify(this.history[this.currentIndex]));
      this.notify();
    }
  }
}
```
