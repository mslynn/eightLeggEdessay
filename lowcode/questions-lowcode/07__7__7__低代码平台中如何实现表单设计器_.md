低代码平台的表单设计器需要考虑以下几个核心功能:

### 3. 表单验证引擎

```javascript
class FormValidator {
  constructor(rules) {
    this.rules = rules;
  }
  
  async validate(value, fieldId) {
    const fieldRules = this.rules[fieldId] || [];
    const errors = [];
    
    for (const rule of fieldRules) {
      const error = await this.validateRule(value, rule);
      if (error) {
        errors.push(error);
      }
    }
    
    return errors;
  }
  
  async validateRule(value, rule) {
    switch (rule.type) {
      case 'required':
        if (!value || value.trim() === '') {
          return rule.message;
        }
        break;
      case 'pattern':
        if (!new RegExp(rule.value).test(value)) {
          return rule.message;
        }
        break;
      case 'email':
        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
          return rule.message;
        }
        break;
      case 'min':
        if (value.length < rule.value) {
          return rule.message;
        }
        break;
      case 'max':
        if (value.length > rule.value) {
          return rule.message;
        }
        break;
    }
  }
}
```

### 4. 表单联动

```javascript
// 字段联动
{
  "id": "city",
  "type": "Select",
  "label": "城市",
  "events": {
    "onChange": "loadDistricts($event)"
  }
}

// 加载区县
function loadDistricts(cityId) {
  return fetch(`/api/districts?cityId=${cityId}`)
    .then(res => res.json())
    .then(districts => {
      // 更新区县下拉框的选项
      updateFieldOptions('district', districts);
    });
}
```

### 5. 表单布局

支持多种布局方式:

```javascript
// 垂直布局
{
  "layout": {
    "type": "vertical",
    "labelWidth": 120,
    "labelAlign": "right"
  }
}

// 水平布局
{
  "layout": {
    "type": "horizontal",
    "columns": 2
  }
}

// 栅格布局
{
  "layout": {
    "type": "grid",
    "columns": 24,
    "gutter": 16
  }
}

// 响应式布局
{
  "layout": {
    "type": "responsive",
    "breakpoints": {
      "xs": { "columns": 1 },
      "sm": { "columns": 2 },
      "md": { "columns": 3 },
      "lg": { "columns": 4 }
    }
  }
}
```
