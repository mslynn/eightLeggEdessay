## 14. 低代码平台中如何实现数据源管理?

**答案:**

数据源管理是低代码平台连接外部数据的核心功能,需要支持多种数据源类型、数据绑定、数据转换等。

### 1. 数据源类型定义

```javascript
const DataSourceTypes = {
  API: 'api',                    // REST API
  GRAPHQL: 'graphql',            // GraphQL
  DATABASE: 'database',          // 数据库
  MOCK: 'mock',                  // Mock数据
  STATIC: 'static',              // 静态数据
  WEBSOCKET: 'websocket'         // WebSocket
};

const DataSourceMethods = {
  GET: 'GET',
  POST: 'POST',
  PUT: 'PUT',
  DELETE: 'DELETE',
  PATCH: 'PATCH'
};
```

### 2. 数据源定义

```javascript
class DataSource {
  constructor(data) {
    this.id = data.id;
    this.name = data.name;
    this.type = data.type;
    this.config = data.config;
    this.transform = data.transform;
    this.cache = data.cache || {};
    this.errorHandler = data.errorHandler;
  }
  
  // 执行数据源
  async execute(context = {}) {
    try {
      let data;
      
      switch (this.type) {
        case DataSourceTypes.API:
          data = await this.executeApi(context);
          break;
        case DataSourceTypes.GRAPHQL:
          data = await this.executeGraphQL(context);
          break;
        case DataSourceTypes.MOCK:
          data = this.executeMock(context);
          break;
        case DataSourceTypes.STATIC:
          data = this.config.data;
          break;
        default:
          throw new Error(`不支持的数据源类型: ${this.type}`);
      }
      
      // 数据转换
      if (this.transform) {
        data = this.transform(data, context);
      }
      
      return data;
    } catch (error) {
      if (this.errorHandler) {
        return this.errorHandler(error, context);
      }
      throw error;
    }
  }
  
  // 执行API请求
  async executeApi(context) {
    const { method = DataSourceMethods.GET, url, headers, params, body } = this.config;
    
    // 替换URL中的变量
    const finalUrl = this.replaceVariables(url, context);
    
    // 替换参数中的变量
    const finalParams = this.replaceVariables(params, context);
    
    const options = {
      method,
      headers: {
        'Content-Type': 'application/json',
        ...headers
      }
    };
    
    if (body && method !== DataSourceMethods.GET) {
      options.body = JSON.stringify(this.replaceVariables(body, context));
    }
    
    // 构建查询参数
    const queryString = new URLSearchParams(finalParams).toString();
    const finalUrlWithQuery = queryString ? `${finalUrl}?${queryString}` : finalUrl;
    
    const response = await fetch(finalUrlWithQuery, options);
    
    if (!response.ok) {
      throw new Error(`请求失败: ${response.status} ${response.statusText}`);
    }
    
    return response.json();
  }
  
  // 执行GraphQL请求
  async executeGraphQL(context) {
    const { url, query, variables } = this.config;
    
    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        query: this.replaceVariables(query, context),
        variables: this.replaceVariables(variables, context)
      })
    });
    
    return response.json();
  }
  
  // 执行Mock数据
  executeMock(context) {
    const { data } = this.config;
    return this.replaceVariables(data, context);
  }
  
  // 替换变量
  replaceVariables(obj, context) {
    if (typeof obj === 'string') {
      return obj.replace(/\$\{([^}]+)\}/g, (match, path) => {
        return this.getValueByPath(context, path);
      });
    }
    
    if (Array.isArray(obj)) {
      return obj.map(item => this.replaceVariables(item, context));
    }
    
    if (typeof obj === 'object' && obj !== null) {
      const result = {};
      Object.keys(obj).forEach(key => {
        result[key] = this.replaceVariables(obj[key], context);
      });
      return result;
    }
    
    return obj;
  }
  
  // 根据路径获取值
  getValueByPath(obj, path) {
    return path.split('.').reduce((acc, key) => acc?.[key], obj);
  }
}
```

### 3. 数据源管理器

```javascript
class DataSourceManager {
  constructor() {
    this.dataSources = new Map();
    this.cache = new Map();
  }
  
  // 注册数据源
  register(dataSource) {
    this.dataSources.set(dataSource.id, dataSource);
    return dataSource;
  }
  
  // 获取数据源
  get(id) {
    return this.dataSources.get(id);
  }
  
  // 执行数据源
  async execute(id, context = {}) {
    const dataSource = this.get(id);
    if (!dataSource) {
      throw new Error(`数据源 ${id} 不存在`);
    }
    
    // 检查缓存
    if (dataSource.cache.enabled) {
      const cacheKey = this.getCacheKey(id, context);
      const cached = this.cache.get(cacheKey);
      
      if (cached && Date.now() - cached.timestamp < dataSource.cache.ttl) {
        return cached.data;
      }
    }
    
    // 执行数据源
    const data = await dataSource.execute(context);
    
    // 缓存数据
    if (dataSource.cache.enabled) {
      const cacheKey = this.getCacheKey(id, context);
      this.cache.set(cacheKey, {
        data,
        timestamp: Date.now()
      });
    }
    
    return data;
  }
  
  // 获取缓存键
  getCacheKey(id, context) {
    return `${id}_${JSON.stringify(context)}`;
  }
  
  // 清除缓存
  clearCache(id) {
    if (id) {
      for (const key of this.cache.keys()) {
        if (key.startsWith(id)) {
          this.cache.delete(key);
        }
      }
    } else {
      this.cache.clear();
    }
  }
  
  // 批量执行数据源
  async executeBatch(requests) {
    const promises = requests.map(req => 
      this.execute(req.id, req.context)
    );
    return Promise.all(promises);
  }
}
```

### 4. 数据绑定

```javascript
class DataBinder {
  constructor(dataSourceManager) {
    this.dataSourceManager = dataSourceManager;
    this.bindings = new Map();
  }
  
  // 创建数据绑定
  createBinding(elementId, dataSourceId, bindingPath) {
    const binding = {
      elementId,
      dataSourceId,
      bindingPath,
      value: null
    };
    
    this.bindings.set(elementId, binding);
    return binding;
  }
  
  // 更新绑定
  async updateBinding(elementId, context = {}) {
    const binding = this.bindings.get(elementId);
    if (!binding) {
      return;
    }
    
    // 执行数据源
    const data = await this.dataSourceManager.execute(binding.dataSourceId, context);
    
    // 提取绑定路径的值
    const value = this.getValueByPath(data, binding.bindingPath);
    binding.value = value;
    
    // 更新元素
    this.updateElement(elementId, value);
    
    return value;
  }
  
  // 更新所有绑定
  async updateAllBindings(context = {}) {
    const promises = Array.from(this.bindings.keys()).map(elementId => 
      this.updateBinding(elementId, context)
    );
    return Promise.all(promises);
  }
  
  // 根据路径获取值
  getValueByPath(obj, path) {
    if (!path) return obj;
    return path.split('.').reduce((acc, key) => acc?.[key], obj);
  }
  
  // 更新元素
  updateElement(elementId, value) {
    const element = document.querySelector(`[data-binding="${elementId}"]`);
    if (element) {
      if (element.tagName === 'INPUT') {
        element.value = value;
      } else {
        element.textContent = value;
      }
    }
  }
}
```

### 5. 表单数据绑定

```javascript
class FormDataBinder extends DataBinder {
  constructor(dataSourceManager) {
    super(dataSourceManager);
    this.formData = {};
  }
  
  // 初始化表单
  initForm(formId, dataSourceId) {
    this.createBinding(formId, dataSourceId, '');
  }
  
  // 加载表单数据
  async loadForm(formId, context = {}) {
    const binding = this.bindings.get(formId);
    if (!binding) {
      return;
    }
    
    const data = await this.dataSourceManager.execute(binding.dataSourceId, context);
    this.formData = data;
    
    // 填充表单字段
    Object.keys(data).forEach(field => {
      const element = document.querySelector(`[name="${field}"]`);
      if (element) {
        element.value = data[field];
      }
    });
    
    return data;
  }
  
  // 收集表单数据
  collectFormData(formId) {
    const form = document.getElementById(formId);
    const formData = new FormData(form);
    const data = {};
    
    formData.forEach((value, key) => {
      data[key] = value;
    });
    
    this.formData = data;
    return data;
  }
  
  // 提交表单
  async submitForm(formId, submitDataSourceId) {
    const data = this.collectFormData(formId);
    
    return await this.dataSourceManager.execute(submitDataSourceId, {
      data
    });
  }
}
```

### 6. 实时数据更新

```javascript
class RealtimeDataSource extends DataSource {
  constructor(data) {
    super(data);
    this.listeners = [];
    this.pollingTimer = null;
  }
  
  // 启动实时更新
  startPolling(interval = 5000) {
    this.stopPolling();
    
    this.pollingTimer = setInterval(async () => {
      try {
        const data = await this.execute();
        this.notifyListeners(data);
      } catch (error) {
        console.error('实时更新失败:', error);
      }
    }, interval);
  }
  
  // 停止实时更新
  stopPolling() {
    if (this.pollingTimer) {
      clearInterval(this.pollingTimer);
      this.pollingTimer = null;
    }
  }
  
  // 添加监听器
  addListener(listener) {
    this.listeners.push(listener);
  }
  
  // 移除监听器
  removeListener(listener) {
    const index = this.listeners.indexOf(listener);
    if (index !== -1) {
      this.listeners.splice(index, 1);
    }
  }
  
  // 通知监听器
  notifyListeners(data) {
    this.listeners.forEach(listener => listener(data));
  }
}
```
