## 15. 低代码平台中如何实现国际化?

**答案:**

国际化(i18n)是低代码平台支持多语言的重要功能,需要支持界面文本、组件标签、数据格式等的多语言切换。

### 2. 国际化管理器

```javascript
class I18nManager {
  constructor() {
    this.currentLocale = 'zh-CN';
    this.locales = new Map();
    this.fallbackLocale = 'zh-CN';
    this.listeners = [];
  }
  
  // 注册语言包
  registerLocale(locale, messages) {
    this.locales.set(locale, messages);
  }
  
  // 设置当前语言
  setLocale(locale) {
    if (!this.locales.has(locale)) {
      console.warn(`语言包 ${locale} 不存在`);
      return;
    }
    
    this.currentLocale = locale;
    this.notifyListeners();
  }
  
  // 获取当前语言
  getLocale() {
    return this.currentLocale;
  }
  
  // 翻译文本
  t(key, params = {}) {
    const messages = this.locales.get(this.currentLocale);
    const fallbackMessages = this.locales.get(this.fallbackLocale);
    
    // 支持嵌套key,如'common.confirm'
    const keys = key.split('.');
    let value = messages;
    
    // 尝试从当前语言包获取
    for (const k of keys) {
      if (value && typeof value === 'object') {
        value = value[k];
      } else {
        value = undefined;
        break;
      }
    }
    
    // 如果当前语言包没有,尝试从fallback语言包获取
    if (value === undefined) {
      value = fallbackMessages;
      for (const k of keys) {
        if (value && typeof value === 'object') {
          value = value[k];
        } else {
          value = key;
          break;
        }
      }
    }
    
    // 替换参数
    if (typeof value === 'string' && params) {
      value = value.replace(/\{(\w+)\}/g, (match, param) => {
        return params[param] !== undefined ? params[param] : match;
      });
    }
    
    return value || key;
  }
  
  // 添加语言变化监听器
  addListener(listener) {
    this.listeners.push(listener);
    return () => {
      const index = this.listeners.indexOf(listener);
      if (index !== -1) {
        this.listeners.splice(index, 1);
      }
    };
  }
  
  // 通知监听器
  notifyListeners() {
    this.listeners.forEach(listener => listener(this.currentLocale));
  }
  
  // 格式化日期
  formatDate(date, format = 'YYYY-MM-DD') {
    const locale = this.currentLocale;
    const options = this.getDateOptions(format);
    return new Intl.DateTimeFormat(locale, options).format(new Date(date));
  }
  
  // 获取日期格式化选项
  getDateOptions(format) {
    const options = {};
    
    if (format.includes('YYYY')) {
      options.year = 'numeric';
    }
    if (format.includes('MM')) {
      options.month = '2-digit';
    }
    if (format.includes('DD')) {
      options.day = '2-digit';
    }
    if (format.includes('HH')) {
      options.hour = '2-digit';
    }
    if (format.includes('mm')) {
      options.minute = '2-digit';
    }
    
    return options;
  }
  
  // 格式化数字
  formatNumber(number, options = {}) {
    const locale = this.currentLocale;
    return new Intl.NumberFormat(locale, options).format(number);
  }
  
  // 格式化货币
  formatCurrency(amount, currency = 'CNY') {
    const locale = this.currentLocale;
    return new Intl.NumberFormat(locale, {
      style: 'currency',
      currency
    }).format(amount);
  }
}
```

### 3. React集成

```javascript
// 创建I18n上下文
const I18nContext = React.createContext();

// I18n提供者组件
export const I18nProvider = ({ children, i18n }) => {
  const [locale, setLocale] = React.useState(i18n.getLocale());
  
  React.useEffect(() => {
    const unsubscribe = i18n.addListener((newLocale) => {
      setLocale(newLocale);
    });
    
    return unsubscribe;
  }, [i18n]);
  
  const value = {
    locale,
    setLocale: (newLocale) => i18n.setLocale(newLocale),
    t: (key, params) => i18n.t(key, params),
    formatDate: (date, format) => i18n.formatDate(date, format),
    formatNumber: (number, options) => i18n.formatNumber(number, options),
    formatCurrency: (amount, currency) => i18n.formatCurrency(amount, currency)
  };
  
  return (
    <I18nContext.Provider value={value}>
      {children}
    </I18nContext.Provider>
  );
};

// 使用I18n的Hook
export const useI18n = () => {
  const context = React.useContext(I18nContext);
  if (!context) {
    throw new Error('useI18n must be used within I18nProvider');
  }
  return context;
};

// 翻译组件
export const T = ({ key, params, ...props }) => {
  const { t } = useI18n();
  return <span {...props}>{t(key, params)}</span>;
};

// 使用示例
function MyComponent() {
  const { t, locale, setLocale, formatDate } = useI18n();
  
  return (
    <div>
      <h1>{t('editor.pageTitle')}</h1>
      <button onClick={() => setLocale('zh-CN')}>中文</button>
      <button onClick={() => setLocale('en-US')}>English</button>
      <p>{formatDate(new Date(), 'YYYY-MM-DD')}</p>
    </div>
  );
}
```

```javascript
// 获取国际化后的组件元数据
function getLocalizedMeta(meta, locale) {
  const localizedMeta = { ...meta };
  
  if (typeof meta.title === 'object') {
    localizedMeta.title = meta.title[locale] || meta.title['zh-CN'];
  }
  
  if (typeof meta.category === 'object') {
    localizedMeta.category = meta.category[locale] || meta.category['zh-CN'];
  }
  
  if (typeof meta.description === 'object') {
    localizedMeta.description = meta.description[locale] || meta.description['zh-CN'];
  }
  
  if (meta.props) {
    localizedMeta.props = meta.props.map(prop => {
      const localizedProp = { ...prop };
      
      if (typeof prop.label === 'object') {
        localizedProp.label = prop.label[locale] || prop.label['zh-CN'];
      }
      
      if (prop.options) {
        localizedProp.options = prop.options.map(option => {
          if (typeof option.label === 'object') {
            return {
              ...option,
              label: option.label[locale] || option.label['zh-CN']
            };
          }
          return option;
        });
      }
      
      return localizedProp;
    });
  }
  
  return localizedMeta;
}
```

### 5. Schema国际化

```javascript
// 支持国际化的Schema
const i18nSchema = {
  version: '1.0.0',
  i18n: {
    'zh-CN': {
      pageTitle: '用户管理',
      username: '用户名',
      email: '邮箱',
      submit: '提交'
    },
    'en-US': {
      pageTitle: 'User Management',
      username: 'Username',
      email: 'Email',
      submit: 'Submit'
    }
  },
  components: [
    {
      type: 'Text',
      props: {
        text: '${i18n.pageTitle}'
      }
    },
    {
      type: 'Input',
      props: {
        label: '${i18n.username}',
        placeholder: '${i18n.username}'
      }
    }
  ]
};

// 解析Schema中的国际化文本
function parseI18nSchema(schema, locale) {
  const parsedSchema = JSON.parse(JSON.stringify(schema));
  
  function parseValue(value) {
    if (typeof value === 'string' && value.startsWith('${i18n.')) {
      const key = value.substring(7, value.length - 1);
      return schema.i18n[locale][key] || key;
    }
    
    if (Array.isArray(value)) {
      return value.map(parseValue);
    }
    
    if (typeof value === 'object' && value !== null) {
      const result = {};
      Object.keys(value).forEach(k => {
        result[k] = parseValue(value[k]);
      });
      return result;
    }
    
    return value;
  }
  
  if (parsedSchema.components) {
    parsedSchema.components = parsedSchema.components.map(comp => {
      if (comp.props) {
        comp.props = parseValue(comp.props);
      }
      return comp;
    });
  }
  
  return parsedSchema;
}
```
