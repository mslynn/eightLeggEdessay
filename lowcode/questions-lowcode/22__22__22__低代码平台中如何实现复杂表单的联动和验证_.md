## 22. 低代码平台中如何实现复杂表单的联动和验证?

**答案:**

复杂表单是低代码平台的重要组成部分,需要支持字段联动、动态验证、条件显示等功能。

### 1. 表单联动实现

```javascript
// 表单联动配置
const formLinkageConfig = {
  // 字段联动规则
  rules: [
    {
      type: 'valueChange',
      sourceField: 'country',
      targetField: 'city',
      action: 'loadOptions',
      config: {
        api: '/api/cities',
        params: {
          country: '${sourceValue}'
        }
      }
    },
    {
      type: 'valueChange',
      sourceField: 'age',
      targetField: 'isAdult',
      action: 'setValue',
      config: {
        expression: '${sourceValue} >= 18'
      }
    },
    {
      type: 'valueChange',
      sourceField: 'userType',
      targetFields: ['companyName', 'position'],
      action: 'showHide',
      config: {
        condition: '${sourceValue} === "enterprise"',
        show: true
      }
    }
  ]
};

// 表单联动引擎
class FormLinkageEngine {
  constructor() {
    this.rules = [];
    this.formData = {};
    this.listeners = [];
  }
  
  // 添加联动规则
  addRule(rule) {
    this.rules.push(rule);
  }
  
  // 初始化表单数据
  initFormData(data) {
    this.formData = { ...data };
  }
  
  // 字段值变化时触发
  onFieldChange(fieldName, value) {
    // 更新表单数据
    this.formData[fieldName] = value;
    
    // 查找相关的联动规则
    const relatedRules = this.rules.filter(rule => rule.sourceField === fieldName);
    
    // 执行联动规则
    relatedRules.forEach(rule => {
      this.executeRule(rule, value);
    });
    
    // 通知监听器
    this.notifyListeners(fieldName, value);
  }
  
  // 执行联动规则
  executeRule(rule, sourceValue) {
    const context = {
      sourceValue,
      formData: this.formData
    };
    
    switch (rule.action) {
      case 'loadOptions':
        this.loadOptions(rule, context);
        break;
        
      case 'setValue':
        this.setValue(rule, context);
        break;
        
      case 'showHide':
        this.showHide(rule, context);
        break;
        
      case 'enableDisable':
        this.enableDisable(rule, context);
        break;
        
      case 'required':
        this.setRequired(rule, context);
        break;
    }
  }
  
  // 加载选项
  async loadOptions(rule, context) {
    const { api, params } = rule.config;
    
    // 替换参数中的变量
    const finalParams = this.replaceVariables(params, context);
    
    try {
      const response = await fetch(api, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(finalParams)
      });
      
      const options = await response.json();
      
      // 更新目标字段的选项
      this.updateFieldOptions(rule.targetField, options);
    } catch (error) {
      console.error('加载选项失败:', error);
    }
  }
  
  // 设置值
  setValue(rule, context) {
    const { expression } = rule.config;
    const value = this.evaluateExpression(expression, context);
    
    this.updateFieldValue(rule.targetField, value);
  }
  
  // 显示/隐藏
  showHide(rule, context) {
    const { condition, show } = rule.config;
    const shouldShow = this.evaluateExpression(condition, context);
    
    if (shouldShow) {
      this.showField(rule.targetFields, show);
    }
  }
  
  // 启用/禁用
  enableDisable(rule, context) {
    const { condition, enable } = rule.config;
    const shouldEnable = this.evaluateExpression(condition, context);
    
    this.enableField(rule.targetFields, shouldEnable);
  }
  
  // 设置必填
  setRequired(rule, context) {
    const { condition, required } = rule.config;
    const isRequired = this.evaluateExpression(condition, context);
    
    this.setFieldRequired(rule.targetField, isRequired);
  }
  
  // 替换变量
  replaceVariables(obj, context) {
    if (typeof obj === 'string') {
      return obj.replace(/\$\{(\w+)\}/g, (match, key) => {
        return context[key] || match;
      });
    }
    
    if (Array.isArray(obj)) {
      return obj.map(item => this.replaceVariables(item, context));
    }
    
    if (typeof obj === 'object' && obj !== null) {
      const result = {};
      Object.keys(obj).forEach(key => {
        result[key] = this.replaceVariables(obj[key], context);
      });
      return result;
    }
    
    return obj;
  }
  
  // 表达式求值
  evaluateExpression(expression, context) {
    const func = new Function('context', `with(context) { return ${expression}; }`);
    return func(context);
  }
  
  // 更新字段选项
  updateFieldOptions(fieldName, options) {
    const field = this.getFieldElement(fieldName);
    if (field) {
      // 更新下拉框选项
      if (field.tagName === 'SELECT') {
        field.innerHTML = options.map(opt => 
          `<option value="${opt.value}">${opt.label}</option>`
        ).join('');
      }
    }
  }
  
  // 更新字段值
  updateFieldValue(fieldName, value) {
    const field = this.getFieldElement(fieldName);
    if (field) {
      field.value = value;
      this.formData[fieldName] = value;
    }
  }
  
  // 显示/隐藏字段
  showField(fieldNames, show) {
    fieldNames.forEach(fieldName => {
      const field = this.getFieldElement(fieldName);
      const fieldWrapper = field?.closest('.form-item');
      
      if (fieldWrapper) {
        fieldWrapper.style.display = show ? '' : 'none';
      }
    });
  }
  
  // 启用/禁用字段
  enableField(fieldNames, enable) {
    fieldNames.forEach(fieldName => {
      const field = this.getFieldElement(fieldName);
      if (field) {
        field.disabled = !enable;
      }
    });
  }
  
  // 设置字段必填
  setFieldRequired(fieldName, required) {
    const field = this.getFieldElement(fieldName);
    if (field) {
      field.required = required;
    }
  }
  
  // 获取字段元素
  getFieldElement(fieldName) {
    return document.querySelector(`[name="${fieldName}"]`);
  }
  
  // 添加监听器
  addListener(listener) {
    this.listeners.push(listener);
  }
  
  // 通知监听器
  notifyListeners(fieldName, value) {
    this.listeners.forEach(listener => {
      listener(fieldName, value, this.formData);
    });
  }
}
```

### 2. 复杂验证规则

```javascript
// 验证规则配置
const validationRules = {
  username: [
    { type: 'required', message: '用户名不能为空' },
    { type: 'minLength', value: 4, message: '用户名至少4个字符' },
    { type: 'maxLength', value: 20, message: '用户名最多20个字符' },
    { type: 'pattern', value: '^[a-zA-Z0-9_]+$', message: '用户名只能包含字母、数字和下划线' }
  ],
  password: [
    { type: 'required', message: '密码不能为空' },
    { type: 'minLength', value: 8, message: '密码至少8个字符' },
    { 
      type: 'custom',
      validator: (value) => /[A-Z]/.test(value) && /[a-z]/.test(value) && /[0-9]/.test(value),
      message: '密码必须包含大小写字母和数字'
    }
  ],
  confirmPassword: [
    { type: 'required', message: '请确认密码' },
    { 
      type: 'custom',
      validator: (value, formData) => value === formData.password,
      message: '两次密码输入不一致'
    }
  ],
  email: [
    { type: 'required', message: '邮箱不能为空' },
    { type: 'email', message: '请输入正确的邮箱格式' }
  ],
  age: [
    { type: 'required', message: '年龄不能为空' },
    { type: 'min', value: 0, message: '年龄不能小于0' },
    { type: 'max', value: 150, message: '年龄不能大于150' }
  ]
};

// 表单验证器
class FormValidator {
  constructor() {
    this.rules = {};
    this.formData = {};
  }
  
  // 添加验证规则
  addRules(fieldName, rules) {
    if (!this.rules[fieldName]) {
      this.rules[fieldName] = [];
    }
    this.rules[fieldName].push(...rules);
  }
  
  // 验证单个字段
  async validateField(fieldName, value) {
    const errors = [];
    const fieldRules = this.rules[fieldName] || [];
    
    for (const rule of fieldRules) {
      const error = await this.validateRule(value, rule, this.formData);
      if (error) {
        errors.push(error);
      }
    }
    
    return errors;
  }
  
  // 验证整个表单
  async validateForm(formData) {
    this.formData = formData;
    const allErrors = {};
    
    for (const fieldName in this.rules) {
      const errors = await this.validateField(fieldName, formData[fieldName]);
      if (errors.length > 0) {
        allErrors[fieldName] = errors;
      }
    }
    
    return {
      valid: Object.keys(allErrors).length === 0,
      errors: allErrors
    };
  }
  
  // 验证单个规则
  async validateRule(value, rule, formData) {
    switch (rule.type) {
      case 'required':
        if (!value || (typeof value === 'string' && value.trim() === '')) {
          return rule.message;
        }
        break;
        
      case 'minLength':
        if (value && value.length < rule.value) {
          return rule.message;
        }
        break;
        
      case 'maxLength':
        if (value && value.length > rule.value) {
          return rule.message;
        }
        break;
        
      case 'min':
        if (value !== undefined && value < rule.value) {
          return rule.message;
        }
        break;
        
      case 'max':
        if (value !== undefined && value > rule.value) {
          return rule.message;
        }
        break;
        
      case 'pattern':
        if (value && !new RegExp(rule.value).test(value)) {
          return rule.message;
        }
        break;
        
      case 'email':
        if (value && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
          return rule.message;
        }
        break;
        
      case 'custom':
        if (rule.validator) {
          const result = await rule.validator(value, formData);
          if (!result) {
            return rule.message;
          }
        }
        break;
        
      case 'async':
        if (rule.validator) {
          const result = await rule.validator(value, formData);
          if (!result.valid) {
            return result.message || rule.message;
          }
        }
        break;
    }
    
    return null;
  }
}
```

### 3. 动态表单渲染

```javascript
// 动态表单配置
const dynamicFormConfig = {
  formId: 'user-form',
  fields: [
    {
      name: 'userType',
      type: 'radio',
      label: '用户类型',
      options: [
        { label: '个人用户', value: 'personal' },
        { label: '企业用户', value: 'enterprise' }
      ],
      defaultValue: 'personal'
    },
    {
      name: 'username',
      type: 'input',
      label: '用户名',
      placeholder: '请输入用户名',
      rules: [
        { type: 'required', message: '用户名不能为空' }
      ]
    },
    {
      name: 'companyName',
      type: 'input',
      label: '公司名称',
      placeholder: '请输入公司名称',
      visible: '${userType === "enterprise"}',
      rules: [
        { 
          type: 'required',
          message: '公司名称不能为空',
          condition: '${userType === "enterprise"}'
        }
      ]
    },
    {
      name: 'position',
      type: 'select',
      label: '职位',
      placeholder: '请选择职位',
      options: [
        { label: '产品经理', value: 'pm' },
        { label: '技术经理', value: 'tm' },
        { label: '销售经理', value: 'sm' }
      ],
      visible: '${userType === "enterprise"}'
    }
  ]
};

// 动态表单渲染器
class DynamicFormRenderer {
  constructor(container, config) {
    this.container = container;
    this.config = config;
    this.formData = {};
    this.components = {
      input: this.renderInput.bind(this),
      select: this.renderSelect.bind(this),
      radio: this.renderRadio.bind(this),
      checkbox: this.renderCheckbox.bind(this),
      date: this.renderDate.bind(this)
    };
  }
  
  // 渲染表单
  render() {
    const form = document.createElement('form');
    form.className = 'dynamic-form';
    
    this.config.fields.forEach(field => {
      const formItem = this.renderFormItem(field);
      form.appendChild(formItem);
    });
    
    this.container.appendChild(form);
    
    // 初始化表单数据
    this.initFormData();
  }
  
  // 渲染表单项
  renderFormItem(field) {
    const item = document.createElement('div');
    item.className = 'form-item';
    item.dataset.fieldName = field.name;
    
    // 检查可见性
    const visible = this.evaluateExpression(field.visible, this.formData);
    item.style.display = visible ? '' : 'none';
    
    // 标签
    const label = document.createElement('label');
    label.className = 'form-label';
    label.textContent = field.label;
    item.appendChild(label);
    
    // 字段组件
    const componentRenderer = this.components[field.type];
    if (componentRenderer) {
      const component = componentRenderer(field);
      item.appendChild(component);
    }
    
    // 错误提示
    const error = document.createElement('div');
    error.className = 'form-error';
    item.appendChild(error);
    
    return item;
  }
  
  // 渲染输入框
  renderInput(field) {
    const input = document.createElement('input');
    input.type = 'text';
    input.name = field.name;
    input.className = 'form-input';
    input.placeholder = field.placeholder;
    input.value = field.defaultValue || '';
    
    input.addEventListener('change', (e) => {
      this.handleFieldChange(field.name, e.target.value);
    });
    
    return input;
  }
  
  // 渲染下拉框
  renderSelect(field) {
    const select = document.createElement('select');
    select.name = field.name;
    select.className = 'form-select';
    
    // 默认选项
    const defaultOption = document.createElement('option');
    defaultOption.value = '';
    defaultOption.textContent = field.placeholder;
    select.appendChild(defaultOption);
    
    // 选项
    field.options.forEach(opt => {
      const option = document.createElement('option');
      option.value = opt.value;
      option.textContent = opt.label;
      select.appendChild(option);
    });
    
    if (field.defaultValue) {
      select.value = field.defaultValue;
    }
    
    select.addEventListener('change', (e) => {
      this.handleFieldChange(field.name, e.target.value);
    });
    
    return select;
  }
  
  // 渲染单选框
  renderRadio(field) {
    const container = document.createElement('div');
    container.className = 'radio-group';
    
    field.options.forEach((opt, index) => {
      const radio = document.createElement('input');
      radio.type = 'radio';
      radio.name = field.name;
      radio.value = opt.value;
      radio.id = `${field.name}-${index}`;
      
      if (opt.value === field.defaultValue) {
        radio.checked = true;
      }
      
      radio.addEventListener('change', (e) => {
        this.handleFieldChange(field.name, e.target.value);
      });
      
      const label = document.createElement('label');
      label.htmlFor = `${field.name}-${index}`;
      label.textContent = opt.label;
      
      const wrapper = document.createElement('div');
      wrapper.appendChild(radio);
      wrapper.appendChild(label);
      
      container.appendChild(wrapper);
    });
    
    return container;
  }
  
  // 字段变化处理
  handleFieldChange(fieldName, value) {
    this.formData[fieldName] = value;
    
    // 更新表单显示
    this.updateFormDisplay();
  }
  
  // 更新表单显示
  updateFormDisplay() {
    this.config.fields.forEach(field => {
      if (field.visible) {
        const item = this.container.querySelector(`[data-field-name="${field.name}"]`);
        if (item) {
          const visible = this.evaluateExpression(field.visible, this.formData);
          item.style.display = visible ? '' : 'none';
        }
      }
    });
  }
  
  // 初始化表单数据
  initFormData() {
    this.config.fields.forEach(field => {
      if (field.defaultValue !== undefined) {
        this.formData[field.name] = field.defaultValue;
      }
    });
  }
  
  // 表达式求值
  evaluateExpression(expression, context) {
    if (!expression) return true;
    
    try {
      const func = new Function('context', `with(context) { return ${expression}; }`);
      return func(context);
    } catch (error) {
      console.error('表达式求值失败:', error);
      return true;
    }
  }
  
  // 获取表单数据
  getFormData() {
    const form = this.container.querySelector('form');
    const formData = new FormData(form);
    const data = {};
    
    formData.forEach((value, key) => {
      data[key] = value;
    });
    
    return data;
  }
}
```
