## 24. 低代码平台如何实现测试策略和质量保证?

**答案:**

低代码平台的测试需要覆盖DSL验证、组件测试、集成测试、端到端测试等多个层面。

### 1. DSL Schema验证

```javascript
// Schema验证器
class SchemaValidator {
  constructor() {
    this.validators = [];
    this.componentRegistry = new Map();
  }
  
  // 注册组件定义
  registerComponent(name, definition) {
    this.componentRegistry.set(name, definition);
  }
  
  // 添加验证规则
  addValidator(validator) {
    this.validators.push(validator);
  }
  
  // 验证Schema
  validate(schema) {
    const errors = [];
    const warnings = [];
    
    // 基础验证
    this.validateBase(schema, errors, warnings);
    
    // 组件验证
    this.validateComponents(schema, errors, warnings);
    
    // 自定义验证器
    this.validators.forEach(validator => {
      const result = validator.validate(schema);
      errors.push(...result.errors);
      warnings.push(...result.warnings);
    });
    
    return {
      valid: errors.length === 0,
      errors,
      warnings
    };
  }
  
  // 基础验证
  validateBase(schema, errors, warnings) {
    if (!schema.version) {
      errors.push({
        type: 'missing-version',
        message: 'Schema缺少版本号'
      });
    }
    
    if (!schema.pageId) {
      errors.push({
        type: 'missing-page-id',
        message: 'Schema缺少页面ID'
      });
    }
    
    if (!schema.components || schema.components.length === 0) {
      warnings.push({
        type: 'empty-components',
        message: 'Schema不包含任何组件'
      });
    }
  }
  
  // 组件验证
  validateComponents(schema, errors, warnings) {
    const componentIds = new Set();
    
    schema.components.forEach(comp => {
      // 验证组件ID
      if (!comp.id) {
        errors.push({
          type: 'missing-component-id',
          message: '组件缺少ID',
          component: comp
        });
      } else if (componentIds.has(comp.id)) {
        errors.push({
          type: 'duplicate-component-id',
          message: `组件ID重复: ${comp.id}`,
          component: comp
        });
      } else {
        componentIds.add(comp.id);
      }
      
      // 验证组件类型
      if (!comp.type) {
        errors.push({
          type: 'missing-component-type',
          message: '组件缺少类型',
          component: comp
        });
      } else if (!this.componentRegistry.has(comp.type)) {
        warnings.push({
          type: 'unknown-component-type',
          message: `未知的组件类型: ${comp.type}`,
          component: comp
        });
      }
      
      // 验证组件属性
      this.validateComponentProps(comp, errors, warnings);
      
      // 递归验证子组件
      if (comp.children) {
        this.validateComponents({ components: comp.children }, errors, warnings);
      }
    });
  }
  
  // 验证组件属性
  validateComponentProps(comp, errors, warnings) {
    const definition = this.componentRegistry.get(comp.type);
    
    if (!definition || !definition.propsSchema) {
      return;
    }
    
    const propsSchema = definition.propsSchema;
    const props = comp.props || {};
    
    // 验证必填属性
    if (propsSchema.required) {
      propsSchema.required.forEach(propName => {
        if (props[propName] === undefined) {
          errors.push({
            type: 'missing-required-prop',
            message: `缺少必填属性: ${propName}`,
            component: comp,
            property: propName
          });
        }
      });
    }
    
    // 验证属性类型
    if (propsSchema.properties) {
      Object.keys(props).forEach(propName => {
        const propValue = props[propName];
        const propSchema = propsSchema.properties[propName];
        
        if (propSchema) {
          const valid = this.validatePropType(propValue, propSchema);
          if (!valid) {
            errors.push({
              type: 'invalid-prop-type',
              message: `属性类型不匹配: ${propName}`,
              component: comp,
              property: propName,
              expected: propSchema.type,
              actual: typeof propValue
            });
          }
        }
      });
    }
  }
  
  // 验证属性类型
  validatePropType(value, schema) {
    switch (schema.type) {
      case 'string':
        return typeof value === 'string';
      case 'number':
        return typeof value === 'number';
      case 'boolean':
        return typeof value === 'boolean';
      case 'array':
        return Array.isArray(value);
      case 'object':
        return typeof value === 'object' && value !== null && !Array.isArray(value);
      default:
        return true;
    }
  }
}

// 使用示例
const validator = new SchemaValidator();

// 注册组件定义
validator.registerComponent('Button', {
  propsSchema: {
    type: 'object',
    properties: {
      text: { type: 'string' },
      type: { type: 'string', enum: ['primary', 'default'] },
      disabled: { type: 'boolean' }
    },
    required: ['text']
  }
});

// 验证Schema
const schema = {
  version: '1.0.0',
  pageId: 'page-001',
  components: [
    {
      id: 'button-001',
      type: 'Button',
      props: {
        type: 'primary'
      }
    }
  ]
};

const result = validator.validate(schema);
console.log(result);
```

### 2. 组件单元测试

```javascript
// 组件测试框架
class ComponentTester {
  constructor() {
    this.testCases = [];
    this.beforeEach = null;
    this.afterEach = null;
  }
  
  // 添加测试用例
  test(name, fn) {
    this.testCases.push({ name, fn });
  }
  
  // 设置前置钩子
  beforeEach(fn) {
    this.beforeEach = fn;
  }
  
  // 设置后置钩子
  afterEach(fn) {
    this.afterEach = fn;
  }
  
  // 运行测试
  async run() {
    const results = [];
    
    for (const testCase of this.testCases) {
      try {
        if (this.beforeEach) {
          await this.beforeEach();
        }
        
        await testCase.fn();
        
        results.push({
          name: testCase.name,
          status: 'passed'
        });
      } catch (error) {
        results.push({
          name: testCase.name,
          status: 'failed',
          error: error.message
        });
      } finally {
        if (this.afterEach) {
          await this.afterEach();
        }
      }
    }
    
    return results;
  }
}

// 测试示例
const tester = new ComponentTester();

// 渲染组件
function renderComponent(componentDef) {
  const container = document.createElement('div');
  // 实际渲染逻辑
  return container;
}

// 查找元素
function querySelector(container, selector) {
  return container.querySelector(selector);
}

// 触发事件
function fireEvent(element, eventType, eventData) {
  const event = new Event(eventType, { bubbles: true });
  Object.assign(event, eventData);
  element.dispatchEvent(event);
}

// 测试用例
tester.test('Button组件应该正确渲染', () => {
  const container = renderComponent({
    type: 'Button',
    props: { text: '点击我', type: 'primary' }
  });
  
  const button = querySelector(container, 'button');
  
  if (!button) {
    throw new Error('Button元素不存在');
  }
  
  if (button.textContent !== '点击我') {
    throw new Error('Button文本不正确');
  }
  
  if (!button.classList.contains('primary')) {
    throw new Error('Button样式不正确');
  }
});

tester.test('Button点击应该触发onClick事件', () => {
  let clicked = false;
  
  const container = renderComponent({
    type: 'Button',
    props: { 
      text: '点击我',
      onClick: () => { clicked = true; }
    }
  });
  
  const button = querySelector(container, 'button');
  fireEvent(button, 'click');
  
  if (!clicked) {
    throw new Error('onClick事件未触发');
  }
});

// 运行测试
tester.run().then(results => {
  console.log('测试结果:', results);
});
```

### 3. 集成测试

```javascript
// 集成测试框架
class IntegrationTester {
  constructor() {
    this.testCases = [];
    this.mockApis = new Map();
  }
  
  // 模拟API
  mockApi(url, response) {
    this.mockApis.set(url, response);
  }
  
  // 添加测试用例
  test(name, fn) {
    this.testCases.push({ name, fn });
  }
  
  // 运行测试
  async run() {
    const results = [];
    
    // 拦截fetch
    const originalFetch = window.fetch;
    window.fetch = async (url, options) => {
      const mockResponse = this.mockApis.get(url);
      
      if (mockResponse) {
        return {
          ok: true,
          json: async () => mockResponse
        };
      }
      
      return originalFetch(url, options);
    };
    
    // 运行测试
    for (const testCase of this.testCases) {
      try {
        await testCase.fn();
        
        results.push({
          name: testCase.name,
          status: 'passed'
        });
      } catch (error) {
        results.push({
          name: testCase.name,
          status: 'failed',
          error: error.message
        });
      }
    }
    
    // 恢复fetch
    window.fetch = originalFetch;
    
    return results;
  }
}

// 测试示例
const integrationTester = new IntegrationTester();

// 模拟API
integrationTester.mockApi('/api/users', [
  { id: 1, name: '张三' },
  { id: 2, name: '李四' }
]);

integrationTester.mockApi('/api/users/1', {
  id: 1,
  name: '张三',
  email: 'zhangsan@example.com'
});

// 测试用例
integrationTester.test('应该正确加载用户列表', async () => {
  const schema = {
    type: 'Table',
    props: {
      dataSource: {
        url: '/api/users'
      }
    }
  };
  
  const container = renderComponent(schema);
  
  // 等待数据加载
  await new Promise(resolve => setTimeout(resolve, 100));
  
  const rows = container.querySelectorAll('tbody tr');
  
  if (rows.length !== 2) {
    throw new Error(`期望2行数据,实际${rows.length}行`);
  }
});

integrationTester.test('点击行应该显示用户详情', async () => {
  const schema = {
    type: 'Table',
    props: {
      dataSource: {
        url: '/api/users'
      }
    },
    events: {
      onRowClick: 'showUserDetail($event.id)'
    }
  };
  
  const container = renderComponent(schema);
  
  await new Promise(resolve => setTimeout(resolve, 100));
  
  const firstRow = container.querySelector('tbody tr');
  fireEvent(firstRow, 'click');
  
  const detailPanel = container.querySelector('.user-detail');
  
  if (!detailPanel) {
    throw new Error('详情面板未显示');
  }
});

// 运行测试
integrationTester.run().then(results => {
  console.log('集成测试结果:', results);
});
```

### 4. 端到端测试

```javascript
// E2E测试框架
class E2ETester {
  constructor() {
    this.browser = null;
    this.page = null;
  }
  
  // 初始化浏览器
  async init() {
    // 使用Puppeteer或Playwright
    const puppeteer = require('puppeteer');
    this.browser = await puppeteer.launch({
      headless: false
    });
    this.page = await this.browser.newPage();
  }
  
  // 导航到页面
  async goto(url) {
    await this.page.goto(url);
  }
  
  // 点击元素
  async click(selector) {
    await this.page.click(selector);
  }
  
  // 输入文本
  async type(selector, text) {
    await this.page.type(selector, text);
  }
  
  // 等待元素
  async waitForSelector(selector) {
    await this.page.waitForSelector(selector);
  }
  
  // 获取文本
  async getText(selector) {
    return await this.page.$eval(selector, el => el.textContent);
  }
  
  // 截图
  async screenshot(path) {
    await this.page.screenshot({ path });
  }
  
  // 关闭浏览器
  async close() {
    await this.browser.close();
  }
}

// 测试示例
async function runE2ETest() {
  const tester = new E2ETester();
  
  try {
    // 初始化
    await tester.init();
    
    // 打开页面
    await tester.goto('http://localhost:3000/lowcode-editor');
    
    // 拖拽组件
    await tester.click('[data-component-type="Button"]');
    await tester.click('#canvas');
    
    // 配置属性
    await tester.type('[name="text"]', '提交按钮');
    await tester.click('[name="type"]');
    await tester.click('[value="primary"]');
    
    // 预览
    await tester.click('#preview-button');
    
    // 验证
    await tester.waitForSelector('button.primary');
    const buttonText = await tester.getText('button.primary');
    
    if (buttonText !== '提交按钮') {
      throw new Error('按钮文本不正确');
    }
    
    // 截图
    await tester.screenshot('preview.png');
    
    console.log('E2E测试通过');
  } catch (error) {
    console.error('E2E测试失败:', error);
  } finally {
    await tester.close();
  }
}
```

### 5. 自动化测试流水线

```javascript
// 测试流水线
class TestPipeline {
  constructor() {
    this.stages = [];
  }
  
  // 添加阶段
  addStage(name, fn) {
    this.stages.push({ name, fn });
  }
  
  // 运行流水线
  async run() {
    const results = {
      stages: [],
      overallStatus: 'passed'
    };
    
    for (const stage of this.stages) {
      const startTime = Date.now();
      
      try {
        await stage.fn();
        
        results.stages.push({
          name: stage.name,
          status: 'passed',
          duration: Date.now() - startTime
        });
      } catch (error) {
        results.stages.push({
          name: stage.name,
          status: 'failed',
          duration: Date.now() - startTime,
          error: error.message
        });
        
        results.overallStatus = 'failed';
      }
    }
    
    return results;
  }
}

// 构建测试流水线
const pipeline = new TestPipeline();

pipeline.addStage('Schema验证', async () => {
  const validator = new SchemaValidator();
  const result = validator.validate(schema);
  
  if (!result.valid) {
    throw new Error(`Schema验证失败: ${JSON.stringify(result.errors)}`);
  }
});

pipeline.addStage('组件单元测试', async () => {
  const tester = new ComponentTester();
  const results = await tester.run();
  
  const failed = results.filter(r => r.status === 'failed');
  if (failed.length > 0) {
    throw new Error(`${failed.length}个单元测试失败`);
  }
});

pipeline.addStage('集成测试', async () => {
  const tester = new IntegrationTester();
  const results = await tester.run();
  
  const failed = results.filter(r => r.status === 'failed');
  if (failed.length > 0) {
    throw new Error(`${failed.length}个集成测试失败`);
  }
});

pipeline.addStage('E2E测试', async () => {
  await runE2ETest();
});

// 运行流水线
pipeline.run().then(results => {
  console.log('测试流水线结果:', results);
  
  if (results.overallStatus === 'passed') {
    console.log('所有测试通过');
  } else {
    console.error('测试失败');
    process.exit(1);
  }
});
```
