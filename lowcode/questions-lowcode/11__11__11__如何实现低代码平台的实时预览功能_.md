## 11. 如何实现低代码平台的实时预览功能?

**答案:**

实时预览是低代码平台的核心功能,让用户在设计时能够即时看到页面效果。

### 1. 预览模式设计

```javascript
const PreviewModes = {
  DESIGN: 'design',    // 设计模式:显示辅助线、选中状态
  PREVIEW: 'preview',  // 预览模式:隐藏编辑相关UI
  PREVIEW_PC: 'preview-pc',  // PC预览
  PREVIEW_MOBILE: 'preview-mobile'  // 移动端预览
};
```

### 2. iframe隔离方案

```javascript
class PreviewEngine {
  constructor(container) {
    this.container = container;
    this.iframe = null;
    this.port = null;
  }
  
  // 创建预览iframe
  createIframe() {
    this.iframe = document.createElement('iframe');
    this.iframe.style.width = '100%';
    this.iframe.style.height = '100%';
    this.iframe.style.border = 'none';
    this.container.appendChild(this.iframe);
    
    // 等待iframe加载完成
    return new Promise((resolve) => {
      this.iframe.onload = () => {
        // 建立通信通道
        this.port = this.iframe.contentWindow;
        this.setupMessageChannel();
        resolve();
      };
      
      // 加载预览页面
      this.iframe.src = '/preview.html';
    });
  }
  
  // 建立消息通道
  setupMessageChannel() {
    // 监听来自iframe的消息
    window.addEventListener('message', (event) => {
      if (event.source === this.port) {
        this.handleMessage(event.data);
      }
    });
  }
  
  // 发送消息到iframe
  sendMessage(type, payload) {
    if (this.port) {
      this.port.postMessage({ type, payload }, '*');
    }
  }
  
  // 更新预览内容
  updatePreview(schema) {
    this.sendMessage('UPDATE_SCHEMA', schema);
  }
  
  // 处理来自iframe的消息
  handleMessage(data) {
    switch (data.type) {
      case 'READY':
        console.log('预览iframe已就绪');
        break;
      case 'COMPONENT_CLICK':
        // 通知主应用组件被点击
        this.emit('component-click', data.payload);
        break;
      case 'COMPONENT_HOVER':
        // 通知主应用组件被悬停
        this.emit('component-hover', data.payload);
        break;
    }
  }
}
```

### 3. 预览页面(iframe内部)

```html
<!-- preview.html -->
<!DOCTYPE html>
<html>
<head>
  <title>预览</title>
  <script src="/preview-renderer.js"></script>
</head>
<body>
  <div id="root"></div>
  <script>
    // 监听来自父窗口的消息
    window.addEventListener('message', (event) => {
      const { type, payload } = event.data;
      
      switch (type) {
        case 'UPDATE_SCHEMA':
          renderSchema(payload);
          break;
      }
    });
    
    // 通知父窗口已就绪
    window.parent.postMessage({ type: 'READY' }, '*');
    
    // 渲染Schema
    function renderSchema(schema) {
      const renderer = new PreviewRenderer();
      renderer.render(schema, document.getElementById('root'));
    }
  </script>
</body>
</html>
```

### 4. 实时渲染

```javascript
class PreviewRenderer {
  constructor() {
    this.schema = null;
    this.componentMap = {};
  }
  
  // 注册组件
  registerComponent(type, component) {
    this.componentMap[type] = component;
  }
  
  // 渲染Schema
  render(schema, container) {
    this.schema = schema;
    container.innerHTML = '';
    
    // 递归渲染组件
    schema.components.forEach(comp => {
      const element = this.renderComponent(comp);
      container.appendChild(element);
    });
  }
  
  // 渲染组件
  renderComponent(component) {
    const Component = this.componentMap[component.type];
    
    if (!Component) {
      console.warn(`未找到组件: ${component.type}`);
      return document.createElement('div');
    }
    
    // 创建DOM元素
    const element = Component.createElement(component);
    
    // 处理子组件
    if (component.children) {
      component.children.forEach(child => {
        const childElement = this.renderComponent(child);
        element.appendChild(childElement);
      });
    }
    
    // 添加事件监听
    if (component.events) {
      Object.keys(component.events).forEach(eventName => {
        element.addEventListener(eventName, (e) => {
          const handler = new Function('e', component.events[eventName]);
          handler(e);
          
          // 通知父窗口
          window.parent.postMessage({
            type: 'COMPONENT_EVENT',
            payload: {
              componentId: component.id,
              eventName,
              data: e
            }
          }, '*');
        });
      });
    }
    
    // 添加悬停事件
    element.addEventListener('mouseenter', () => {
      window.parent.postMessage({
        type: 'COMPONENT_HOVER',
        payload: { componentId: component.id }
      }, '*');
    });
    
    element.addEventListener('click', (e) => {
      e.stopPropagation();
      window.parent.postMessage({
        type: 'COMPONENT_CLICK',
        payload: { componentId: component.id }
      }, '*');
    });
    
    return element;
  }
}
```

### 5. 热更新优化

```javascript
// 使用Diff算法进行增量更新
function diffAndRender(oldSchema, newSchema, container) {
  const diff = computeDiff(oldSchema, newSchema);
  
  diff.changes.forEach(change => {
    switch (change.type) {
      case 'ADD':
        // 添加新组件
        const newElement = renderComponent(change.component);
        container.appendChild(newElement);
        break;
        
      case 'REMOVE':
        // 移除组件
        const elementToRemove = container.querySelector(`[data-id="${change.id}"]`);
        if (elementToRemove) {
          elementToRemove.remove();
        }
        break;
        
      case 'UPDATE':
        // 更新组件
        const elementToUpdate = container.querySelector(`[data-id="${change.id}"]`);
        if (elementToUpdate) {
          updateComponent(elementToUpdate, change.component);
        }
        break;
        
      case 'MOVE':
        // 移动组件
        const elementToMove = container.querySelector(`[data-id="${change.id}"]`);
        const targetParent = container.querySelector(`[data-id="${change.targetId}"]`);
        if (elementToMove && targetParent) {
          targetParent.appendChild(elementToMove);
        }
        break;
    }
  });
}
```

### 6. 响应式预览

```javascript
// 移动端预览
class MobilePreview {
  constructor(container) {
    this.container = container;
    this.wrapper = null;
    this.deviceSize = {
      width: 375,
      height: 667
    };
  }
  
  // 创建移动端预览容器
  createMobilePreview() {
    this.wrapper = document.createElement('div');
    this.wrapper.style.cssText = `
      width: ${this.deviceSize.width}px;
      height: ${this.deviceSize.height}px;
      margin: 0 auto;
      border: 1px solid #ddd;
      border-radius: 20px;
      overflow: hidden;
      position: relative;
    `;
    
    this.container.appendChild(this.wrapper);
    
    return this.wrapper;
  }
  
  // 切换设备尺寸
  setDeviceSize(width, height) {
    this.deviceSize = { width, height };
    this.wrapper.style.width = `${width}px`;
    this.wrapper.style.height = `${height}px`;
  }
}
```
