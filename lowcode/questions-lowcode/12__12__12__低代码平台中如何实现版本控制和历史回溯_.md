## 12. 低代码平台中如何实现版本控制和历史回溯?

**答案:**

版本控制和历史回溯是低代码平台的重要功能,允许用户保存快照、比较差异、回退到历史版本。

### 1. 版本数据结构

```javascript
class Version {
  constructor(data) {
    this.id = generateId();
    this.versionNumber = data.versionNumber;
    this.schema = data.schema;
    this.snapshot = data.snapshot;
    this.createdAt = new Date();
    this.createdBy = data.createdBy;
    this.description = data.description;
    this.tags = data.tags || [];
  }
}
```

### 2. 版本管理器

```javascript
class VersionManager {
  constructor() {
    this.versions = [];
    this.currentVersion = null;
    this.maxVersions = 100;
  }
  
  // 创建新版本
  createVersion(schema, options = {}) {
    const version = new Version({
      versionNumber: this.versions.length + 1,
      schema: schema,
      snapshot: this.createSnapshot(schema),
      createdBy: options.createdBy || 'anonymous',
      description: options.description || '',
      tags: options.tags || []
    });
    
    this.versions.push(version);
    this.currentVersion = version;
    
    // 限制版本数量
    if (this.versions.length > this.maxVersions) {
      this.versions.shift();
    }
    
    return version;
  }
  
  // 创建快照
  createSnapshot(schema) {
    // 使用JSON序列化创建快照
    return JSON.stringify(schema);
  }
  
  // 获取当前版本
  getCurrentVersion() {
    return this.currentVersion;
  }
  
  // 获取所有版本
  getAllVersions() {
    return this.versions;
  }
  
  // 根据ID获取版本
  getVersionById(id) {
    return this.versions.find(v => v.id === id);
  }
  
  // 回退到指定版本
  revertToVersion(versionId) {
    const version = this.getVersionById(versionId);
    if (!version) {
      throw new Error(`版本 ${versionId} 不存在`);
    }
    
    // 恢复Schema
    this.currentVersion = version;
    return version.schema;
  }
  
  // 比较两个版本的差异
  compareVersions(versionId1, versionId2) {
    const v1 = this.getVersionById(versionId1);
    const v2 = this.getVersionById(versionId2);
    
    if (!v1 || !v2) {
      throw new Error('版本不存在');
    }
    
    // 计算差异
    const diff = this.computeDiff(v1.schema, v2.schema);
    return diff;
  }
  
  // 计算Schema差异
  computeDiff(schema1, schema2) {
    const diff = {
      added: [],
      removed: [],
      modified: []
    };
    
    // 比较组件
    const components1 = this.indexComponents(schema1.components);
    const components2 = this.indexComponents(schema2.components);
    
    // 查找新增的组件
    Object.keys(components2).forEach(id => {
      if (!components1[id]) {
        diff.added.push(components2[id]);
      }
    });
    
    // 查找删除的组件
    Object.keys(components1).forEach(id => {
      if (!components2[id]) {
        diff.removed.push(components1[id]);
      }
    });
    
    // 查找修改的组件
    Object.keys(components2).forEach(id => {
      if (components1[id] && JSON.stringify(components1[id]) !== JSON.stringify(components2[id])) {
        diff.modified.push({
          id,
          old: components1[id],
          new: components2[id]
        });
      }
    });
    
    return diff;
  }
  
  // 索引组件
  indexComponents(components) {
    const index = {};
    
    function traverse(comps) {
      comps.forEach(comp => {
        index[comp.id] = comp;
        if (comp.children) {
          traverse(comp.children);
        }
      });
    }
    
    traverse(components);
    return index;
  }
}
```

### 3. 本地存储持久化

```javascript
class PersistentVersionManager extends VersionManager {
  constructor(storageKey, storage = localStorage) {
    super();
    this.storageKey = storageKey;
    this.storage = storage;
    this.loadVersions();
  }
  
  // 保存到本地存储
  saveVersions() {
    try {
      const data = {
        versions: this.versions,
        currentVersionId: this.currentVersion?.id
      };
      this.storage.setItem(this.storageKey, JSON.stringify(data));
    } catch (error) {
      console.error('保存版本失败:', error);
    }
  }
  
  // 从本地存储加载
  loadVersions() {
    try {
      const data = this.storage.getItem(this.storageKey);
      if (data) {
        const parsed = JSON.parse(data);
        this.versions = parsed.versions || [];
        this.currentVersion = this.versions.find(v => v.id === parsed.currentVersionId);
      }
    } catch (error) {
      console.error('加载版本失败:', error);
    }
  }
  
  // 重写createVersion
  createVersion(schema, options = {}) {
    const version = super.createVersion(schema, options);
    this.saveVersions();
    return version;
  }
  
  // 重写revertToVersion
  revertToVersion(versionId) {
    const schema = super.revertToVersion(versionId);
    this.saveVersions();
    return schema;
  }
}
```

### 4. 远程存储同步

```javascript
class RemoteVersionManager extends PersistentVersionManager {
  constructor(storageKey, apiEndpoint) {
    super(storageKey);
    this.apiEndpoint = apiEndpoint;
  }
  
  // 同步到远程服务器
  async syncToRemote() {
    try {
      const response = await fetch(`${this.apiEndpoint}/versions`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          versions: this.versions,
          currentVersionId: this.currentVersion?.id
        })
      });
      
      return response.json();
    } catch (error) {
      console.error('同步到远程失败:', error);
      throw error;
    }
  }
  
  // 从远程服务器加载
  async loadFromRemote() {
    try {
      const response = await fetch(`${this.apiEndpoint}/versions`);
      const data = await response.json();
      
      this.versions = data.versions || [];
      this.currentVersion = this.versions.find(v => v.id === data.currentVersionId);
      this.saveVersions();
      
      return data;
    } catch (error) {
      console.error('从远程加载失败:', error);
      throw error;
    }
  }
}
```

### 5. 自动保存

```javascript
class AutoSaveVersionManager extends VersionManager {
  constructor(options = {}) {
    super();
    this.autoSaveInterval = options.autoSaveInterval || 60000; // 默认1分钟
    this.maxAutoSaveVersions = options.maxAutoSaveVersions || 10;
    this.timer = null;
  }
  
  // 启动自动保存
  startAutoSave() {
    this.stopAutoSave();
    this.timer = setInterval(() => {
      this.autoSave();
    }, this.autoSaveInterval);
  }
  
  // 停止自动保存
  stopAutoSave() {
    if (this.timer) {
      clearInterval(this.timer);
      this.timer = null;
    }
  }
  
  // 自动保存
  autoSave() {
    if (this.currentVersion) {
      const version = this.createVersion(this.currentVersion.schema, {
        description: '自动保存',
        tags: ['auto-save']
      });
      
      // 限制自动保存版本数量
      const autoSaveVersions = this.versions.filter(v => v.tags.includes('auto-save'));
      if (autoSaveVersions.length > this.maxAutoSaveVersions) {
        const oldestAutoSave = autoSaveVersions[0];
        const index = this.versions.indexOf(oldestAutoSave);
        this.versions.splice(index, 1);
      }
      
      console.log('自动保存成功:', version.id);
    }
  }
}
```

### 6. 版本对比UI

```javascript
function renderVersionDiff(diff) {
  const container = document.createElement('div');
  container.className = 'version-diff';
  
  // 添加的组件
  if (diff.added.length > 0) {
    const addedSection = document.createElement('div');
    addedSection.className = 'diff-section added';
    addedSection.innerHTML = '<h3>新增组件</h3>';
    
    diff.added.forEach(comp => {
      const item = document.createElement('div');
      item.className = 'diff-item';
      item.textContent = `${comp.type} (${comp.id})`;
      addedSection.appendChild(item);
    });
    
    container.appendChild(addedSection);
  }
  
  // 删除的组件
  if (diff.removed.length > 0) {
    const removedSection = document.createElement('div');
    removedSection.className = 'diff-section removed';
    removedSection.innerHTML = '<h3>删除组件</h3>';
    
    diff.removed.forEach(comp => {
      const item = document.createElement('div');
      item.className = 'diff-item';
      item.textContent = `${comp.type} (${comp.id})`;
      removedSection.appendChild(item);
    });
    
    container.appendChild(removedSection);
  }
  
  // 修改的组件
  if (diff.modified.length > 0) {
    const modifiedSection = document.createElement('div');
    modifiedSection.className = 'diff-section modified';
    modifiedSection.innerHTML = '<h3>修改组件</h3>';
    
    diff.modified.forEach(change => {
      const item = document.createElement('div');
      item.className = 'diff-item';
      item.textContent = `${change.id}`;
      
      // 显示具体变化
      const details = document.createElement('div');
      details.className = 'diff-details';
      details.innerHTML = `
        <div class="diff-old">${JSON.stringify(change.old)}</div>
        <div class="diff-new">${JSON.stringify(change.new)}</div>
      `;
      
      item.appendChild(details);
      modifiedSection.appendChild(item);
    });
    
    container.appendChild(modifiedSection);
  }
  
  return container;
}
```
