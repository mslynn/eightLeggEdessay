# 1. 富文本编辑器的复制粘贴功能是如何实现的？有更好的实现方式吗？

**答案：**

富文本编辑器的复制粘贴功能主要通过以下方式实现：

#### 1. 使用 Clipboard API

```javascript
// 复制功能
async function copyRichText() {
  try {
    const htmlContent = '<b>粗体</b> 和 <i>斜体</i> 文本';
    const textContent = '粗体 和 斜体 文本';

    await navigator.clipboard.write([
      new ClipboardItem({
        'text/html': new Blob([htmlContent], { type: 'text/html' }),
        'text/plain': new Blob([textContent], { type: 'text/plain' })
      })
    ]);
  } catch (err) {
    console.error('复制失败:', err);
  }
}

// 粘贴功能
async function pasteRichText() {
  try {
    const clipboardItems = await navigator.clipboard.read();
    
    for (const item of clipboardItems) {
      if (item.types.includes('text/html')) {
        const htmlBlob = await item.getType('text/html');
        const htmlText = await htmlBlob.text();
        return htmlText;
      }
    }
  } catch (err) {
    console.error('粘贴失败:', err);
  }
}
```

#### 2. 监听粘贴事件

document.addEventListener('paste', async (event) => {

#### 3. 使用惰性函数优化复制粘贴性能

```javascript
// 惰性函数：只在第一次调用时确定具体实现，后续直接使用
const getClipboardCopy = function() {
  // 检测浏览器支持
  const supportsClipboardAPI = !!navigator.clipboard && typeof navigator.clipboard.write === 'function';
  const supportsExecCommand = !!document.execCommand;
  
  // 返回优化后的复制函数
  if (supportsClipboardAPI) {
    return function(htmlContent, textContent) {
      return navigator.clipboard.write([
        new ClipboardItem({
          'text/html': new Blob([htmlContent], { type: 'text/html' }),
          'text/plain': new Blob([textContent], { type: 'text/plain' })
        })
      ]);
    };
  } else if (supportsExecCommand) {
    return function(htmlContent, textContent) {
      // 创建临时元素
      const textarea = document.createElement('textarea');
      textarea.style.position = 'fixed';
      textarea.style.opacity = '0';
      textarea.value = htmlContent;
      document.body.appendChild(textarea);
      textarea.select();
      
      try {
        const success = document.execCommand('copy');
        document.body.removeChild(textarea);
        return success ? Promise.resolve() : Promise.reject(new Error('复制失败'));
      } catch (err) {
        document.body.removeChild(textarea);
        return Promise.reject(err);
      }
    };
  } else {
    return function() {
      return Promise.reject(new Error('浏览器不支持复制功能'));
    };
  }
}();

// 使用惰性函数的复制
async function copyRichTextOptimized() {
  const htmlContent = '<b>粗体</b> 和 <i>斜体</i> 文本';
  const textContent = '粗体 和 斜体 文本';

// 惰性函数：粘贴功能
const getClipboardPaste = function() {
  const supportsClipboardAPI = !!navigator.clipboard && typeof navigator.clipboard.read === 'function';
  const supportsPasteEvent = typeof ClipboardEvent !== 'undefined';
  
  if (supportsClipboardAPI) {
    return async function() {
      try {
        const clipboardItems = await navigator.clipboard.read();
        
        for (const item of clipboardItems) {
          if (item.types.includes('text/html')) {
            const htmlBlob = await item.getType('text/html');
            return await htmlBlob.text();
          }
        }
        
        // 如果没有 HTML，返回纯文本
        const textBlob = await clipboardItems[0].getType('text/plain');
        return await textBlob.text();
      } catch (err) {
        throw new Error('读取剪贴板失败');
      }
    };
  } else if (supportsPasteEvent) {
    return function() {
      return new Promise((resolve, reject) => {
        // 创建一个隐藏的输入框来触发粘贴事件
        const textarea = document.createElement('textarea');
        textarea.style.position = 'fixed';
        textarea.style.opacity = '0';
        document.body.appendChild(textarea);
        textarea.focus();
        
        const handler = (event) => {
          event.preventDefault();
          const html = event.clipboardData.getData('text/html');
          const text = event.clipboardData.getData('text/plain');
          
          document.removeEventListener('paste', handler);
          document.body.removeChild(textarea);
          
          resolve(html || text);
        };
        
        document.addEventListener('paste', handler, { once: true });
        
        // 触发粘贴
        textarea.dispatchEvent(new KeyboardEvent('keydown', {
          key: 'v',
          ctrlKey: true,
          keyCode: 86
        }));
      });
    };
  } else {
    return function() {
      return Promise.reject(new Error('浏览器不支持粘贴功能'));
    };
  }
}();

// 使用惰性函数的粘贴
async function pasteRichTextOptimized() {
  try {
    const content = await getClipboardPaste();
    console.log('粘贴内容:', content);
    return content;
  } catch (err) {
    console.error('粘贴失败:', err);
  }
}
```

**惰性函数的优势：**

1. **性能优化**：只在第一次调用时检测浏览器能力，后续直接使用缓存的实现
2. **兼容性处理**：自动降级到浏览器支持的最佳实现
3. **代码简洁**：封装了复杂的兼容性判断逻辑
4. **易于维护**：浏览器能力检测和实际实现分离

#### 更好的实现方式：

1. **使用 Slate.js 或 Quill**：专业的富文本编辑器框架
2. **使用 ContentEditable + Selection API**：更底层的控制
3. **使用 ProseMirror**：最强大的富文本编辑器框架
4. **使用惰性函数优化**：提升性能和兼容性
