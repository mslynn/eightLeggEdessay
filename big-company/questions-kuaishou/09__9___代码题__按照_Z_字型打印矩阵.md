# 9. 【代码题】按照 Z 字型打印矩阵

**答案：**

```javascript
function printZMatrix(matrix) {
  if (!matrix || matrix.length === 0) {
    return [];
  }

  const rows = matrix.length;
  const cols = matrix[0].length;
  const result = [];
  
  let row = 0;
  let col = 0;
  let goingUp = true;

  while (result.length < rows * cols) {
    result.push(matrix[row][col]);

    if (goingUp) {
      if (col === cols - 1) {
        row++;
        goingUp = false;
      } else if (row === 0) {
        col++;
        goingUp = false;
      } else {
        row--;
        col++;
      }
    } else {
      if (row === rows - 1) {
        col++;
        goingUp = true;
      } else if (col === 0) {
        row++;
        goingUp = true;
      } else {
        row++;
        col--;
      }
    }
  }

  return result;
}

// 测试
console.log(printZMatrix([
  [1, 2, 3],
  [4, 5, 6],
  [7, 8, 9]
])); // [1, 2, 3, 5, 7, 8, 6, 4, 9]
```

**时间复杂度**：O(m × n)
