```javascript
function smoothScroll(targetY) {
  const startY = window.scrollY;
  const distance = targetY - startY;
  const duration = 500;
  let startTime = null;

  function animate(currentTime) {
    if (!startTime) startTime = currentTime;
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);
    
    window.scrollTo(0, startY + distance * progress);
    
    if (progress < 1) {
      requestAnimationFrame(animate);
    }
  }

  requestAnimationFrame(animate);
}
```

**自动化检测**

```javascript
// 1. 内存泄露检测工具
class MemoryLeakDetector {
  constructor() {
    this.snapshots = [];
    this.threshold = 1024 * 1024; // 1MB
  }
  
  async takeSnapshot() {
    const snapshot = await performance.measureUserAgentSpecificMemory();
    this.snapshots.push(snapshot);
    return snapshot;
  }
  
  checkLeak() {
    if (this.snapshots.length < 2) {
      console.log('Need at least 2 snapshots');
      return;
    }
    
    const [first, last] = [
      this.snapshots[0],
      this.snapshots[this.snapshots.length - 1]
    ];
    
    const diff = last.bytes - first.bytes;
    
    if (diff > this.threshold) {
      console.warn(`Possible memory leak: ${(diff / 1024 / 1024).toFixed(2)} MB`);
    }
  }
  
  clear() {
    this.snapshots = [];
  }
}

// 2. 自动化测试
describe('Memory Leak Tests', () => {
  it('should not leak memory after component destruction', async () => {
    const detector = new MemoryLeakDetector();
```

**方案 1: 清理事件监听器**

```javascript
// 使用 AbortController
class Component {
  constructor() {
    this.abortController = new AbortController();
    
    document.addEventListener('click', this.handleClick.bind(this), {
      signal: this.abortController.signal
    });
  }
  
  handleClick() {
    console.log('Clicked');
  }
  
  destroy() {
    // 一次性清理所有事件监听器
    this.abortController.abort();
  }
}

// 使用 WeakMap
const eventListeners = new WeakMap();

function addEventListener(element, event, handler) {
  if (!eventListeners.has(element)) {
    eventListeners.set(element, new Map());
  }
  
  const listeners = eventListeners.get(element);
  if (!listeners.has(event)) {
    listeners.set(event, new Set());
  }
  
  listeners.get(event).add(handler);
  element.addEventListener(event, handler);
}

function removeAllEventListeners(element) {
  const listeners = eventListeners.get(element);
  if (!listeners) return;
  
  listeners.forEach((handlers, event) => {
    handlers.forEach(handler => {
      element.removeEventListener(event, handler);
    });
  });
  
  eventListeners.delete(element);
}
```

**方案 2: 清理定时器**

```javascript
class TimerManager {
  constructor() {
    this.timers = new Set();
    this.rafIds = new Set();
  }
  
  setInterval(callback, delay) {
    const timer = setInterval(callback, delay);
    this.timers.add(timer);
    return timer;
  }
  
  setTimeout(callback, delay) {
    const timer = setTimeout(() => {
      callback();
      this.timers.delete(timer);
    }, delay);
    this.timers.add(timer);
    return timer;
  }
  
  requestAnimationFrame(callback) {
    const rafId = requestAnimationFrame((time) => {
      callback(time);
      this.rafIds.delete(rafId);
    });
    this.rafIds.add(rafId);
    return rafId;
  }
  
  clear() {
    this.timers.forEach(timer => {
      clearInterval(timer);
      clearTimeout(timer);
    });
    this.timers.clear();
    
    this.rafIds.forEach(rafId => {
      cancelAnimationFrame(rafId);
    });
    this.rafIds.clear();
  }
}

// 使用
const timerManager = new TimerManager();
timerManager.setInterval(() => console.log('Tick'), 1000);

// 组件销毁时
timerManager.clear();
```

**方案 4: 清理 DOM 引用**

```javascript
class DOMCache {
  constructor() {
    this.cache = new Map();
  }
  
  get(selector) {
    if (this.cache.has(selector)) {
      const element = this.cache.get(selector);
      // 检查元素是否仍在 DOM 中
      if (document.contains(element)) {
        return element;
      } else {
        this.cache.delete(selector);
      }
    }
    
    const element = document.querySelector(selector);
    if (element) {
      this.cache.set(selector, element);
    }
    
    return element;
  }
  
  clear() {
    this.cache.clear();
  }
}
```

```javascript
// 1. 性能指标监控
class PerformanceMonitor {
  constructor() {
    this.metrics = {};
  }
  
  measure(name, fn) {
    const start = performance.now();
    const result = fn();
    const end = performance.now();
    
    this.metrics[name] = end - start;
    return result;
  }
  
  async measureAsync(name, fn) {
    const start = performance.now();
    const result = await fn();
    const end = performance.now();
    
    this.metrics[name] = end - start;
    return result;
  }
  
  getMetrics() {
    return { ...this.metrics };
  }
  
  reset() {
    this.metrics = {};
  }
}
```

**总结：**

前端性能优化需要从多个维度入手：
1. **加载性能**：代码分割、CDN、资源压缩、图片优化
2. **运行性能**：虚拟列表、防抖节流、Web Worker、requestAnimationFrame
3. **渲染性能**：减少重排重绘、使用 transform、优化虚拟 DOM
4. **内存管理**：及时清理事件监听器、定时器、DOM 引用
5. **监控检测**：使用 DevTools、自动化测试、性能监控工具

内存泄露的发现和解决是一个循环过程：发现 → 分析 → 修复 → 验证，需要持续关注和优化。
